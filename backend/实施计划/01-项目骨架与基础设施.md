```ts
export interface AppConfig {
  databaseUrl: string; redisUrl: string; sessionSecret: string; sessionTtl: number;
  orderPendingTtlMinutes: number; jobScanIntervalSeconds: number;
  corsOrigins: string[]; bcryptCost: number; seedDemoData: boolean;
  minio: { endpoint: string; accessKey: string; secretKey: string; bucket: string; publicUrl: string };
}
export function loadConfig(): AppConfig {
  const required = ['DATABASE_URL', 'REDIS_URL', 'SESSION_SECRET'];
  for (const k of required) if (!process.env[k]) throw new Error(`缺少环境变量 ${k}`);
  return { databaseUrl: process.env.DATABASE_URL!, redisUrl: process.env.REDIS_URL!,
    sessionSecret: process.env.SESSION_SECRET!, sessionTtl: Number(process.env.SESSION_TTL ?? 86400),
    orderPendingTtlMinutes: Number(process.env.ORDER_PENDING_TTL_MINUTES ?? 15),
    jobScanIntervalSeconds: Number(process.env.JOB_SCAN_INTERVAL_SECONDS ?? 60),
    corsOrigins: (process.env.CORS_ORIGINS ?? '').split(',').filter(Boolean),
    bcryptCost: Number(process.env.BCRYPT_COST ?? 12),
    seedDemoData: process.env.SEED_DEMO_DATA === 'true',
    minio: { endpoint: process.env.MINIO_ENDPOINT ?? '', accessKey: process.env.MINIO_ACCESS_KEY ?? '',
      secretKey: process.env.MINIO_SECRET_KEY ?? '', bucket: process.env.MINIO_BUCKET ?? 'zhixu',
      publicUrl: process.env.MINIO_PUBLIC_URL ?? '' } };
}
```

`config.module.ts`：`ConfigModule.forRoot({ isGlobal: true, load: [() => loadConfig()] })`。注意 `loadConfig` 需同时可被直接调用（测试与 `main.ts` 用），不要只挂在 DI 上。

- `prisma.service.ts`：`@Injectable() class PrismaService implements OnModuleInit, OnModuleDestroy`（`onModuleInit` 里 `$connect`，**建连失败只告警不抛出**，否则依赖不可用时 `/healthz` 也起不来）；`PrismaModule` 标记 `@Global()`。本机不跑 `prisma generate`（引擎超时）——`@prisma/client` 由 CI 生成并随依赖安装（CI 执行 `prisma:generate` 后提交或发布），本机仅写代码不执行迁移。**懒加载 Proxy**：构造函数返回 `new Proxy(this, …)`，`new PrismaClient()` 推迟到首次真实访问才执行（未 generate 时构造即抛错，若在构造期实例化则本地应用无法启动）；Nest 启动期探测属性（isPromise 的 `then`、`onApplicationBootstrap`/`beforeApplicationShutdown`/`onApplicationShutdown` 钩子扫描、symbol 属性）一律返回 `undefined`，探测本身不得触发实例化（任务 6 冒烟自测发现后补充）；类型上声明合并 `interface PrismaService extends PrismaClient` 保持消费方用法不变。未 generate 时 `PrismaClient` 类型为 `any`，消费方（如 `IdempotencyService`）以窄接口 `IdempotencyStore` 声明所用 delegate，保证本机也有类型检查。

- `lock.service.ts`：`acquire` 用 `setNxEx`，释放时 `del`（值比对防误删）。
- `rate-limit.service.ts`：`INCR` + 首次 `EXPIRE`，超限抛 `new BizException(ErrorCode.RATE_LIMITED, '请求过于频繁，请稍后再试', 429)`（1005 已在任务 3 随框架异常映射表补齐）。
