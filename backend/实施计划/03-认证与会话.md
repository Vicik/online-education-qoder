export function sessionMiddleware(cfg: AppConfig): RequestHandler {
  const RedisStore = connectRedis(require('express-session'));
  return session({
    store: new RedisStore({ client: redisClient(cfg.redisUrl), prefix: 'sess:' }),
    secret: cfg.sessionSecret, resave: false, saveUninitialized: false,
    cookie: { httpOnly: true, secure: process.env.NODE_ENV === 'production',
              sameSite: 'lax', maxAge: cfg.sessionTtl * 1000 },
    rolling: true,
  });
}

- [ ] **2.6 实现 `auth.service.ts` 核心逻辑**

```ts
async register(dto: RegisterDto) {
  if (dto.password !== dto.confirmPassword) throw new BizException(1001, '两次输入的密码不一致', 400);
  const email = dto.email.toLowerCase();                    // 统一小写存储
  try {
    await this.prisma.user.create({ data: { email, nickname: dto.nickname,
      passwordHash: await bcrypt.hash(dto.password, this.cfg.bcryptCost) } });
  } catch (e) { /* P2002 邮箱已存在 */ throw new BizException(1001, '该邮箱已注册', 409); }
  return { id: (await this.prisma.user.findUnique({ where: { email } }))!.id };
}

async login(dto: LoginDto, req: Request) {
  const email = dto.email.toLowerCase();
  await this.rateLimit.check(`login:ip:${req.ip}`, 5, 60);
  await this.rateLimit.check(`login:email:${email}`, 5, 60);
  const user = await this.prisma.user.findUnique({ where: { email } });
  const ok = user && await bcrypt.compare(dto.password, user.passwordHash);
  if (!ok || user!.status !== 'active') throw new BizException(2003, '邮箱或密码错误', 401); // 统一提示
  if (user!.role === 'admin') {                             // D8：管理员强制 TOTP
    if (!dto.totpCode || !user!.totpSecret || !this.totp.verify(user!.totpSecret, dto.totpCode))
      throw new BizException(2003, '邮箱或密码错误', 401);  // 不泄露 TOTP 状态
  }
  await this.sessions.establish(req, user!.id, user!.role);
  return { role: user!.role };
}
```

`SessionService.establish`：`req.session.userId = userId; req.session.role = role;` + `SADD user_sessions:{userId} <sessionId>`（`req.sessionID` 在 `save` 回调后可得，用 `req.session.save(cb)` 内登记）。`destroyUserSessions`：取 `SMEMBERS` 逐个 `DEL sess:{id}` 再 `DEL` 集合。`logout`：销毁当前会话 + `SREM`。`changePassword`：校验 `confirmPassword` 一致、旧密码正确（错误抛 `BizException(1001, '当前密码不正确', 400)`）、bcrypt 新哈希、`update` 后调 `destroyUserSessions`。
`TotpService.issueSecret`：`authenticator.generateSecret()` 落 `totpSecret`，URI 用 `otpauth://totp/知序课堂:{email}?secret=...`；`verify` 用 `authenticator.verify({ token: code, secret })`。
种子衔接：步骤 02 的 seed 已为 admin 打印提示；本步在 seed 中补一行：admin 若无 `totpSecret` 则生成并打印 URI（dev 控制台扫码）。
注册限流：`register` 入口 `rateLimit.check('register:ip:' + req.ip, 10, 3600)`（防批量注册，D7 暂不启用邮箱验证的替代）。
