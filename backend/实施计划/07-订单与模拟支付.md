```ts
// trade/pricing/pricing.types.ts
export interface PriceQuote {
  priceCents: number;                       // 当前价（售价/秒杀价/拼团价）
  activityType: ActivityType;
  activityId: number | null;
}
export interface PricingStrategy {
  readonly activityType: ActivityType;      // 'normal' | 'groupon' | 'flash_sale'
  /** 校验活动有效性并给出当前价；无效抛 BizException(422 语义) */
  quote(input: { courseId: number; activityId: number | null }): Promise<PriceQuote>;
}
export const PRICING_STRATEGIES = Symbol('PRICING_STRATEGIES');

// trade/pricing/pricing.service.ts
class PricingService {
  /** 注册表查询策略；未知 activityType 抛 1001 */
  resolve(activityType: ActivityType): PricingStrategy;
  /** 预览：策略价 + 券校验（本人、未使用、适用课程、在有效期）→ CheckoutPreviewData */
  preview(userId: number, courseId: number, q: CheckoutQuery): Promise<CheckoutPreviewData>;
}

// trade/payment/post-payment-handler.interface.ts
export interface PostPaymentHandler {
  readonly activityType: ActivityType;
  /** 支付成功事务内调用；抛错则整个支付事务回滚 */
  onPaid(tx: Prisma.TransactionClient, order: Order): Promise<void>;
}
export const POST_PAYMENT_HANDLERS = Symbol('POST_PAYMENT_HANDLERS');

// trade/payment/payment-gateway.interface.ts
export interface PaymentGateway {
  /** 模拟支付恒成功；真实渠道实现预留 */
  charge(order: Order): Promise<{ transactionNo: string }>;
  /** 退款（拼团失败由步骤 12 调用）；真实渠道预留 */
  refund(order: Order, payment: Payment): Promise<{ transactionNo: string }>;
}

// trade/order/order.service.ts
class OrderService {
  /** 创建订单：校验已发布/未重复购买/活动有效/券可用 → 服务端定价 → 事务（order + 核销券 + purchaseKey） */
  createOrder(userId: number, dto: CreateOrderDto, idempotencyKey: string): Promise<Order>;
  /** 支付：幂等；事务（order→paid + payments + PostPaymentHandler）；已支付返回首次结果 */
  pay(userId: number, orderId: number): Promise<Order>;
  getOrder(userId: number, orderId: number): Promise<Order>;       // 归属校验
  listUserOrders(userId: number, page: PageQuery): Promise<PageResult<Order>>;
  listAllOrders(page: PageQuery): Promise<PageResult<Order>>;      // admin
  /** 【供步骤 08/09 复用】创建活动订单的内部入口（幂等键由调用方传） */
  createOrderWithQuote(userId: number, quote: PriceQuote, dto: CreateOrderDto, idempotencyKey: string): Promise<Order>;
}
```

```ts
async createOrder(userId, dto, idempotencyKey) {
  return this.idempotency.run(userId, idempotencyKey, 'POST /orders', async () => {
    const quote = await this.pricing.resolve(dto.activityType ?? 'normal')
      .quote({ courseId: dto.courseId, activityId: dto.activityId ?? null });
    return this.createOrderWithQuote(userId, quote, dto, idempotencyKey);
  });
}

async createOrderWithQuote(userId, quote, dto, idempotencyKey) {
  const course = await this.prisma.course.findUnique({ where: { id: dto.courseId } });
  if (!course || course.status !== 'published') throw new BizException(1001, '课程已下架，不可购买', 422);
  const coupon = dto.couponClaimId
    ? await this.prisma.couponClaim.findUnique({ where: { id: dto.couponClaimId } }) : null;
  const discountCents = coupon?.template.discountCents ?? 0;   // 券额（预览同源校验在 preview；此处再校验归属/状态）
  const paidCents = Math.max(0, quote.priceCents - discountCents);
  const orderNo = `ZX${dayjs().format('YYYYMMDDHHmmss')}${rand4()}`;
  const order = await this.prisma.$transaction(async (tx) => {
    try {
      const created = await tx.order.create({ data: {
        orderNo, userId, courseId: dto.courseId,
        activityType: quote.activityType, activityId: quote.activityId,
        originalCents: course.originalCents, currentCents: quote.priceCents,
        discountCents, paidCents, couponClaimId: dto.couponClaimId ?? null,
        purchaseKey: `user:${userId}:course:${dto.courseId}`,
        flashPurchaseKey: quote.activityType === 'flash_sale'
          ? `user:${userId}:flash:${quote.activityId}` : null,
        expiresAt: new Date(Date.now() + cfg.orderPendingTtlMinutes * 60_000),
      } });
      if (coupon) await this.coupons.redeemInTx(tx, coupon.id, userId, created.id); // 核销：used + 挂订单
      return created;
    } catch (e: any) {
      if (e.code === 'P2002') throw new BizException(4003, '您已购买过该课程', 409);
      throw e;
    }
  });
  return order;
}
```

> 券可用性（归属/未使用/适用课程/在有效期）在创建事务前完整校验一次（`couponClaim` 联查 `template.courses` + 时间），不满足抛 `4002`；事务内 `redeemInTx` 完成「置 used + 写 orderId」（步骤 06 实现为 `updateMany({ where: { id, userId, status: 'unused' } })` 条件更新 + 挂单，计数 0 抛 `4002`，防止并发双单核销同一张券）。
> 防重复购买由 `purchaseKey` 唯一约束兜底（含并发）；purchaseKey 在退款/取消时置 NULL（步骤 12）。

```ts
async pay(userId, orderId) {
  const order = await this.prisma.order.findUnique({ where: { id: orderId } });
  if (!order || order.userId !== userId) throw new BizException(2002, '无权访问该订单', 403);
  if (order.status === 'paid') return order;                      // 幂等：返回首次结果
  if (order.status !== 'pending') throw new BizException(1001, '订单不可支付', 422);
  const { transactionNo } = await this.gateway.charge(order);     // mock 恒成功
  return this.prisma.$transaction(async (tx) => {
    const paid = await tx.order.update({ where: { id: orderId }, data: { status: 'paid', paidAt: new Date() } });
    await tx.payment.create({ data: { orderId, channel: 'mock', transactionNo, amountCents: order.paidCents } });
    const handler = this.handlers.find(h => h.activityType === order.activityType)!;
    await handler.onPaid(tx, paid);                               // 抛错 → 全部回滚
    return paid;
  });
}
```

`mock-payment.gateway.ts`：`charge` 返回 `{ transactionNo: 'MOCK-' + 随机 }`；`refund` 返回同构（步骤 12 用）。`payment-gateway.interface.ts` 注释预留 `wechat/alipay` 渠道实现点（`payments.channel` 字段承载）。
路由：`GET /checkout/:courseId/preview`（登录，query：couponId/activityType/activityId）、`POST /orders`（登录 + Idempotency-Key）、`POST /orders/:id/pay`（登录）、`GET /orders/:id`（登录）、`GET /user/orders`（登录）、`GET /admin/orders`（admin，含用户信息联查）。
