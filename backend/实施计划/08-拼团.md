```ts
async createGroup(userId, activityId, idempotencyKey) {
  return this.idempotency.run(userId, idempotencyKey, 'POST /groupons/groups', async () => {
    const act = await this.prisma.grouponActivity.findUnique({ where: { id: activityId } });
    if (!act || act.status !== 'ongoing' || act.endAt < new Date())
      throw new BizException(5002, '拼团活动未开始或已结束', 422);
    const order = await this.orders.createOrderWithQuote(userId,
      await this.pricing.resolve('groupon').quote({ courseId: act.courseId, activityId }),
      { courseId: act.courseId, activityType: 'groupon', activityId: null }, idempotencyKey);
    return this.prisma.$transaction(async (tx) => {
      const group = await tx.grouponGroup.create({ data: { activityId, status: 'pending',
        memberCount: 0, expiresAt: act.endAt } });
      await tx.grouponMember.create({ data: { groupId: group.id, activityId, userId, orderId: order.id } });
      return { group, order };
    });
  });
}

async joinGroup(userId, activityId, groupId, idempotencyKey) {
  return this.idempotency.run(userId, idempotencyKey, 'POST /groupons/join', async () => {
    const group = await this.prisma.grouponGroup.findUnique({ where: { id: groupId } });
    if (!group || group.activityId !== activityId || group.status !== 'pending')
      throw new BizException(5001, '该团已满或已结束', 422);
    const act = await this.prisma.grouponActivity.findUnique({ where: { id: activityId } });
    if (!act || act.status !== 'ongoing' || act.endAt < new Date())
      throw new BizException(5002, '拼团活动未开始或已结束', 422);
    if (group.memberCount >= act.targetSize) throw new BizException(5001, '该团已满', 422);
    const order = await this.orders.createOrderWithQuote(userId,
      await this.pricing.resolve('groupon').quote({ courseId: act.courseId, activityId }),
      { courseId: act.courseId, activityType: 'groupon', activityId: null }, idempotencyKey);
    try {
      await this.prisma.grouponMember.create({ data: { groupId, activityId, userId, orderId: order.id } });
    } catch (e: any) {
      if (e.code === 'P2002') throw new BizException(5001, '您已参与过该拼团活动', 409);
      throw e;
    }
    return { group, order };
  });
}
```

> 注意：`order.activityId` 在成员/团组建立后回写（order.create 时 activityId=null，成团流程靠 `groupon_members.order_id` 关联团组——`memberCount` 行锁在支付时计数）。`activityId` 回写：`order.update({ data: { activityId: group.id } })` 于事务内执行（保持 schema 注释「groupon: 团组 id」）。

```ts
// groupon-post-payment.handler.ts
export class GrouponPostPaymentHandler implements PostPaymentHandler {
  readonly activityType = 'groupon' as const;
  async onPaid(tx: Prisma.TransactionClient, order: Order) {
    if (order.activityType !== 'groupon') return;
    const member = await tx.grouponMember.findUnique({ where: { orderId: order.id } });
    if (!member) return;                                    // 非拼团订单防御
    // 团组行锁（§9.2）：并发支付串行化
    await tx.$queryRaw`SELECT id FROM groupon_groups WHERE id = ${member.groupId} FOR UPDATE`;
    const group = await tx.grouponGroup.findUnique({ where: { id: member.groupId } });
    const next = group!.memberCount + 1;
    const full = next >= (await tx.grouponActivity.findUnique({ where: { id: member.activityId } }))!.targetSize;
    await tx.grouponGroup.update({ where: { id: member.groupId },
      data: { memberCount: next, ...(full ? { status: 'success', formedAt: new Date() } : {}) } });
  }
}
```

注册：`PromotionModule` 中 `{ provide: POST_PAYMENT_HANDLERS, useClass: GrouponPostPaymentHandler, multi: true }`、`{ provide: PRICING_STRATEGIES, useClass: GrouponPricingStrategy, multi: true }`。
`GrouponPricingStrategy.quote`：`activityId` 为团组 id 时校验团组 `pending` 且未满；给出 `priceCents = activity.priceCents`；活动非 `ongoing` 抛 422。
