class FlashStockService {
  /** 预热：活动开始时（Worker 调用）SET 剩余 = stock - sold */
  warmup(activityId: number, remaining: number): Promise<void>;
  /** 预扣：DECR；<0 回补并返回 false（售罄） */
  preDeduct(activityId: number): Promise<boolean>;
  /** 回补：订单取消/超时/支付失败时 INCR */
  restore(activityId: number): Promise<void>;
  /** 剩余库存：优先 Redis，缺失回源 DB（stock - sold） */
  remaining(activityId: number): Promise<number>;
  /** 键名常量 */
  readonly KEY_PREFIX = 'flash:stock:';
}

```ts
export class FlashStockService {
  constructor(private readonly redis: RedisService, private readonly prisma: PrismaService) {}
  async warmup(activityId: number, remaining: number) { await this.redis.set(`${this.KEY_PREFIX}${activityId}`, String(remaining)); }
  async preDeduct(activityId: number): Promise<boolean> {
    const v = await this.redis.decr(`${this.KEY_PREFIX}${activityId}`);
    if (v < 0) { await this.redis.incr(`${this.KEY_PREFIX}${activityId}`); return false; }
    return true;
  }
  async restore(activityId: number) { await this.redis.incr(`${this.KEY_PREFIX}${activityId}`); }
  async remaining(activityId: number): Promise<number> {
    const v = await this.redis.get(`${this.KEY_PREFIX}${activityId}`);
    if (v !== null) return Number(v);
    const act = await this.prisma.flashSaleActivity.findUnique({ where: { id: activityId } });
    if (!act) throw new BizException(3001, '秒杀活动不存在', 404);
    await this.redis.set(`${this.KEY_PREFIX}${activityId}`, String(act.stock - act.sold));
    return act.stock - act.sold;
  }
}
```

```ts
async purchase(userId, activityId, idempotencyKey) {
  return this.idempotency.run(userId, idempotencyKey, 'POST /flash-sales/purchase', async () => {
    const act = await this.prisma.flashSaleActivity.findUnique({ where: { id: activityId } });
    if (!act) throw new BizException(3001, '秒杀活动不存在', 404);
    if (act.status !== 'ongoing' || act.endAt < new Date())
      throw new BizException(5002, '秒杀活动未开始或已结束', 422);
    if (!(await this.stock.preDeduct(activityId)))
      throw new BizException(4001, '已售罄', 422);
    try {
      return await this.orders.createOrderWithQuote(userId,
        await this.pricing.resolve('flash_sale').quote({ courseId: act.courseId, activityId }),
        { courseId: act.courseId, activityType: 'flash_sale', activityId }, idempotencyKey);
    } catch (e: any) {
      await this.stock.restore(activityId);           // 下单失败释放预占
      if (e.code === 'P2002' && e.meta?.target?.includes('flashPurchaseKey'))
        throw new BizException(5003, '同一活动限购一次', 409);
      throw e;
    }
  });
}
```

`flash-sale-post-payment.handler.ts`（DB 兜底扣减 §9.1）：

```ts
export class FlashSalePostPaymentHandler implements PostPaymentHandler {
  readonly activityType = 'flash_sale' as const;
  async onPaid(tx: Prisma.TransactionClient, order: Order) {
    if (order.activityType !== 'flash_sale') return;
    const { count } = await tx.flashSaleActivity.updateMany({
      where: { id: order.activityId!, sold: { lt: /* stock 列 */ } }, data: { sold: { increment: 1 } } });
    // Prisma 不支持列间比较：改为两步——行锁读活动 → 校验 sold<stock → update
  }
}
```

> Prisma 列间比较限制：`updateMany({ where: { id, sold: { lt: stock } } })` 不合法。改用「行锁读 → 校验 → 更新」：`SELECT ... FOR UPDATE` 读活动行 → `sold < stock` 不满足则抛 `BizException(4001, '已售罄', 422)`（支付事务回滚，订单保持 pending，Redis 预占由调用方回补——pay 失败路径在 `OrderService.pay` 捕获 handler 异常后 `stock.restore`，见下）→ 满足则 `update({ data: { sold: { increment: 1 } } })`。行锁保证并发支付串行扣减，`sold` 永不超 `stock`。
> `OrderService.pay` 需补一行：捕获 handler 抛错时，若订单类型为 flash_sale，调用 `FlashStockService.restore(order.activityId)` 后再抛出（注入方式：pay 内不依赖 flash 模块——用 `POST_PAYMENT_HANDLERS` 之外新增可选接口 `onPaidFailed`？**简化**：`PostPaymentHandler` 接口增加 `onPaidFailed?(order: Order): Promise<void>`，flash 处理器实现 restore；pay 的 catch 中调用）。

路由：`GET /flash-sales`（公开，列表含剩余库存与倒计时——倒计时字段 `startAt/endAt` ISO UTC，前端按 Asia/Shanghai 渲染）、`GET /flash-sales/:id`（公开，`stock/total`、`remaining`）、`POST /flash-sales/:id/purchase`（登录 + Idempotency-Key）。
