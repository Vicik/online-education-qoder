- [ ] **1.2 运行确认失败**；**1.3 实现**

```ts
export type ImageType = 'png' | 'jpeg' | 'webp';
export function detectImageType(buf: Buffer): ImageType {
  if (buf.length >= 8 && buf.subarray(0, 8).equals(Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]))) return 'png';
  if (buf.length >= 3 && buf[0] === 0xff && buf[1] === 0xd8 && buf[2] === 0xff) return 'jpeg';
  if (buf.length >= 12 && buf.subarray(0, 4).toString('ascii') === 'RIFF' &&
      buf.subarray(8, 12).toString('ascii') === 'WEBP') return 'webp';
  throw new BizException(1001, '仅支持 PNG/JPEG/WebP 图片', 400);
}
```

```ts
async uploadImage(file) {
  if (file.buffer.length > 2 * 1024 * 1024) throw new BizException(1001, '图片不能超过 2MB', 400);
  const type = detectImageType(file.buffer);
  const mimeMap = { png: 'image/png', jpeg: 'image/jpeg', webp: 'image/webp' };
  if (file.mimeType !== mimeMap[type]) throw new BizException(1001, '图片内容与类型不符', 400);
  const cleaned = await sharp(file.buffer).rotate()               // 按 EXIF 方向摆正并丢弃元数据
    [type === 'png' ? 'png' : type === 'jpeg' ? 'jpeg' : 'webp']().toBuffer();
  const key = `zhixu/${yyyy}/${mm}/${randomUUID()}.${type === 'jpeg' ? 'jpg' : type}`;
  await this.s3.send(new PutObjectCommand({ Bucket: cfg.minio.bucket, Key: key,
    Body: cleaned, ContentType: mimeMap[type] }));
  const url = await this.presignGet(key, 900);
  return { key, url };
}
async presignGet(key: string, expiresInSec = 900) {
  const url = await getSignedUrl(this.s3, new GetObjectCommand({
    Bucket: cfg.minio.bucket, Key: key }), { expiresIn: expiresInSec });
  return url;
}
```

`storage.module.ts`：`S3Client` 按 `MINIO_ENDPOINT/ACCESS_KEY/SECRET_KEY` 初始化（`forcePathStyle: true` 兼容 MinIO）。`files.controller.ts`：`POST /files/upload`（登录；`@UseInterceptors(FileInterceptor('file'))` 内存缓冲 + `limits: { fileSize: 2MB }`），fileSize 超限由 multer 抛错 → filter 映射 1001/400。桶不存在时（`NoSuchBucket`）在 module init 自动 `CreateBucketCommand`（幂等）。
