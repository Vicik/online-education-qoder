```ts
describe('E2E 购买链路', () => {
  it('注册→登录→浏览→领券→预览→下单→支付→学习→购买记录', async () => {
    const app = await createTestApp();                    // 真实 DB/Redis（CI 服务容器）
    const a = request.agent(app.getHttpServer());
    await a.post('/api/v1/auth/register').send({ email: 'e2e-buy@test.com', password: 'E2ePassw0rd',
      confirmPassword: 'E2ePassw0rd', nickname: 'E2E 买家' }).expect(201);
    await a.post('/api/v1/auth/login').send({ email: 'e2e-buy@test.com', password: 'E2ePassw0rd' }).expect(200);
    const courses = await a.get('/api/v1/courses').expect(200);
    expect(courses.body.data.items.length).toBeGreaterThan(0);
    const claim = await a.post('/api/v1/coupons/1/claim').set('Idempotency-Key', 'e2e-buy-c1').expect(201);
    const prev = await a.get(`/api/v1/checkout/1/preview?couponId=${claim.body.data.id}`).expect(200);
    const order = await a.post('/api/v1/orders').set('Idempotency-Key', 'e2e-buy-o1')
      .send({ courseId: 1, couponClaimId: claim.body.data.id }).expect(201);
    const paid = await a.post(`/api/v1/orders/${order.body.data.id}/pay`).expect(200);
    expect(paid.body.data.paidCents).toBe(prev.body.data.paidCents);
    await a.get('/api/v1/courses/1/chapters/1/content').expect(200);
    await a.get('/api/v1/courses/1/chapters/1/content');  // 已学完校验略
    const records = await a.get('/api/v1/user/orders').expect(200);
    expect(records.body.data.items.some((o: any) => o.id === order.body.data.id)).toBe(true);
  });
});
```
