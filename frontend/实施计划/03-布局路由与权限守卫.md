```tsx
import type { ComponentType, ReactNode } from 'react';
import { Navigate, useLocation } from 'react-router';
import type { SessionUser } from '@/stores/session';
import { useSessionStore } from '@/stores/session';

// 角色与布局键
export type Role = 'user' | 'admin';
export type LayoutKey = 'auth' | 'user' | 'user-center' | 'admin';

// 懒加载页面
export type LazyPage = () => Promise<{ default: ComponentType }>;

// 路由元信息：权限只在这里声明
export interface RouteMeta {
  path: string;
  layout: LayoutKey;
  requiresAuth: boolean;
  roles?: Role[];
  lazy: LazyPage;
}

// 按优先级判定跳转目标；返回 null 表示放行
export function resolveRedirect(
  user: SessionUser | null,
  meta: RouteMeta,
  path: string,
): string | null {
  // 1. 管理员隔离：非 /admin 路由一律回管理端
  if (user?.role === 'admin' && !path.startsWith('/admin')) return '/admin';
  // 2. 未登录访问受保护路由：带原地址去登录
  if (meta.requiresAuth && !user) return `/login?redirect=${encodeURIComponent(path)}`;
  // 3. 已登录但角色不匹配
  if (meta.roles && user && !meta.roles.includes(user.role)) return '/403';
  return null;
}

// 路由守卫：命中跳转就重定向，否则渲染子节点
export function GuardedRoute({ meta, children }: { meta: RouteMeta; children: ReactNode }) {
  const user = useSessionStore((state) => state.user);
  const location = useLocation();
  const redirect = resolveRedirect(user, meta, `${location.pathname}${location.search}`);
  if (redirect) return <Navigate to={redirect} replace />;
  return <>{children}</>;
}
```

```ts
// 导航项
export interface NavItem {
  path: string;
  label: string;
}

// 用户端顶部导航（考试仅普通用户显示）
const USER_BASE_NAV_ITEMS: NavItem[] = [

  return role === 'user' ? [...USER_BASE_NAV_ITEMS, { path: '/exams', label: '考试' }] : USER_BASE_NAV_ITEMS;

// 用户中心侧栏 9 项

  { path: '/user/coupons', label: '我的优惠券' },
  { path: '/user/favorites', label: '我的收藏' },
  { path: '/user/profile', label: '修改资料' },
  { path: '/user/password', label: '修改密码' },
];

// 管理端侧栏 8 项（无「查看用户端」）
export const ADMIN_NAV_ITEMS: NavItem[] = [
  { path: '/admin', label: '工作台' },
  { path: '/admin/courses', label: '课程管理' },
  { path: '/admin/groupons', label: '拼团管理' },
  { path: '/admin/flash-sales', label: '秒杀管理' },
  { path: '/admin/questions', label: '题库管理' },
  { path: '/admin/exams', label: '考试管理' },
  { path: '/admin/coupons', label: '优惠券' },
  { path: '/admin/blocked-words', label: '屏蔽词' },
];

// 按最长前缀匹配侧栏项；/admin 必须精确匹配
function findSelectedKey(items: NavItem[], pathname: string, rootPath: string): string {
  const matched = items
    .filter((item) => (item.path === rootPath ? pathname === rootPath : pathname.startsWith(item.path)))
    .sort((left, right) => right.path.length - left.path.length)[0];
  return matched?.path ?? rootPath;
}

// 管理端侧栏高亮
export function getAdminSelectedKey(pathname: string): string {
  return findSelectedKey(ADMIN_NAV_ITEMS, pathname, '/admin');
}

// 用户中心侧栏高亮
export function getUserCenterSelectedKey(pathname: string): string {
  return findSelectedKey(USER_CENTER_NAV_ITEMS, pathname, '/user');
}
```

```tsx
import { Button, Dropdown, Layout, Menu } from 'antd';
import { Link, Outlet, useLocation, useNavigate } from 'react-router';
import { getUserNavItems } from '@/app/navigation';
import { useLogout } from '@/hooks/useLogout';
import { useSessionStore } from '@/stores/session';
import './layout.css';

  const user = useSessionStore((state) => state.user);
  const location = useLocation();
  const navigate = useNavigate();
  const logout = useLogout();
  const navItems = getUserNavItems(user?.role ?? null);
```

```ts
import type { components } from './schema';
import { request } from './client';

// 会话用户类型
export type SessionUser = components['schemas']['User'];

// 当前用户；/auth/me 探测不触发全局跳登录
export function getMe(): Promise<SessionUser> {
  return request<SessionUser>({ url: '/auth/me', method: 'GET', skipAuthRedirect: true });
}

// 退出登录
export function logout(): Promise<null> {
  return request<null>({ url: '/auth/logout', method: 'POST' });
}
```

`src/hooks/useLogout.ts`：

```ts
import { useNavigate } from 'react-router';
import { logout } from '@/api/auth';
import { queryClient } from '@/app/query-client';
import { useSessionStore } from '@/stores/session';

// 登出：无论接口成败都清本地会话并回首页
export function useLogout(): () => Promise<void> {
  const clear = useSessionStore((state) => state.clear);
  const navigate = useNavigate();
  return async () => {
    try {
      await logout();
    } finally {
      clear();
      queryClient.clear();
      navigate('/');
    }
  };
}
```

`src/app/session-gate.tsx`：

```tsx
import { useQuery } from '@tanstack/react-query';
import { Spin } from 'antd';
import { useEffect, type PropsWithChildren } from 'react';
import { getMe } from '@/api/auth';
import { queryKeys } from '@/api/query-keys';
import { useSessionStore } from '@/stores/session';

// 启动时探测一次会话；失败按游客处理
export function SessionGate({ children }: PropsWithChildren) {
  const setUser = useSessionStore((state) => state.setUser);
  const query = useQuery({
    queryKey: queryKeys.authMe,
    queryFn: getMe,
    retry: false,
    staleTime: 60_000,
  });

  useEffect(() => {
    setUser(query.data ?? null);
  }, [query.data, setUser]);

  if (query.isLoading) {
    return (
      <div className="full-page-loading">
        <Spin size="large" />
      </div>
    );
  }
  return <>{children}</>;
}
```

```tsx
import { QueryClientProvider } from '@tanstack/react-query';
import { App as AntdApp, ConfigProvider } from 'antd';
import zhCN from 'antd/locale/zh_CN';
import { useEffect, type PropsWithChildren } from 'react';
import { registerAuthHandlers } from '@/api/errors';
import { useSessionStore } from '@/stores/session';
import { APP_THEME_TOKEN } from '@/styles/theme';
import { queryClient } from './query-client';
import { SessionGate } from './session-gate';

// 应用级 Provider：主题、消息上下文、Query、会话引导、全局错误跳转
export function AppProviders({ children }: PropsWithChildren) {
  useEffect(() => {
    registerAuthHandlers({
      onUnauthorized: () => {
        useSessionStore.getState().clear();
        queryClient.clear();
        const current = `${window.location.pathname}${window.location.search}`;
        window.location.assign(`/login?redirect=${encodeURIComponent(current)}`);
      },
      onForbidden: () => window.location.assign('/403'),
    });
  }, []);

  return (
    <ConfigProvider locale={zhCN} theme={{ token: APP_THEME_TOKEN }}>
      <AntdApp>
        <QueryClientProvider client={queryClient}>
          <SessionGate>{children}</SessionGate>
        </QueryClientProvider>
      </AntdApp>
    </ConfigProvider>
  );
}
```
