import type { components } from './schema';
import { request } from './client';

// 会话用户类型
export type SessionUser = components['schemas']['User'];

// 登录请求：管理端额外携带 6 位动态码
export interface LoginPayload {
  email: string;
  password: string;
  totpCode?: string;
}

// 注册请求
export interface RegisterPayload {
  email: string;
  password: string;
  confirmPassword: string;
  nickname: string;
}

// 当前用户；/auth/me 探测不触发全局跳登录
export function getMe(): Promise<SessionUser> {
  return request<SessionUser>({ url: '/auth/me', method: 'GET', skipAuthRedirect: true });
}

// 登录：角色由账号携带，前端不选择角色
export function login(payload: LoginPayload): Promise<SessionUser> {
  return request<SessionUser>({
    url: '/auth/login',
    method: 'POST',
    data: payload,
    skipAuthRedirect: true,
  });
}

// 注册：成功不自动登录
export function register(payload: RegisterPayload): Promise<null> {
  return request<null>({
    url: '/auth/register',
    method: 'POST',
    data: payload,
    skipAuthRedirect: true,
  });
}

// 退出登录
export function logout(): Promise<null> {
  return request<null>({ url: '/auth/logout', method: 'POST' });
}
```

import { Alert, App as AntdApp, Button, Form, Input, Typography } from 'antd';
import { useState } from 'react';
import { Link, useNavigate, useSearchParams } from 'react-router';
import { login } from '@/api/auth';
import { ApiError } from '@/api/errors';
import { queryKeys } from '@/api/query-keys';
import { queryClient } from '@/app/query-client';
import { useSessionStore } from '@/stores/session';
import { validateEmail, validatePassword } from '@/lib/validators';

// 登录表单
interface LoginFormValues {
  email: string;
  password: string;
}

// 登录后的落地地址：管理员固定 /admin；普通用户仅接受站内 redirect
export function resolvePostLoginPath(
  user: { role: 'user' | 'admin' },
  redirect: string | null,
): string {
  if (user.role === 'admin') return '/admin';
  if (redirect && redirect.startsWith('/') && !redirect.startsWith('//')) return redirect;
  return '/user';
}

// 登录页
export default function LoginPage() {
  const [form] = Form.useForm<LoginFormValues>();
  const [searchParams] = useSearchParams();
  const navigate = useNavigate();
  const setUser = useSessionStore((state) => state.setUser);
  const { message } = AntdApp.useApp();
  const [error, setError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState(false);
  const registered = searchParams.get('registered') === '1';
  const passwordChanged = searchParams.get('changed') === '1';

  // 提交登录：成功写入会话并跳转
  async function handleFinish(values: LoginFormValues) {
    setError(null);
    setSubmitting(true);
    try {
      const loggedIn = await login(values);
      queryClient.setQueryData(queryKeys.authMe, loggedIn);
      setUser(loggedIn);
      message.success('登录成功');
      navigate(resolvePostLoginPath(loggedIn, searchParams.get('redirect')), { replace: true });
    } catch (caught) {
      setError(caught instanceof ApiError ? caught.message : '登录失败，请稍后重试');
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <div style={{ padding: 32, background: '#fff', borderRadius: 6, boxShadow: 'var(--shadow-card)' }}>
      <Typography.Title level={3} style={{ marginTop: 0 }}>
        登录
      </Typography.Title>
      {registered && <Alert type="success" showIcon message="注册成功，请登录" style={{ marginBottom: 16 }} />}
      {passwordChanged && <Alert type="success" showIcon message="密码已修改，请重新登录" style={{ marginBottom: 16 }} />}
      {error && <Alert type="error" showIcon message={error} style={{ marginBottom: 16 }} />}
      <Form form={form} layout="vertical" requiredMark={false} onFinish={handleFinish}>
        <Form.Item
          name="email"
          label="邮箱"
          rules={[
            { required: true, message: '请输入邮箱' },
            {
              validator: (_, value: string) => {
                const message = validateEmail(value ?? '');
                return message ? Promise.reject(new Error(message)) : Promise.resolve();
              },
            },
          ]}
        >
          <Input placeholder="user@example.com" autoComplete="email" />
        </Form.Item>
        <Form.Item
          name="password"
          label="密码"
          rules={[
            { required: true, message: '请输入密码' },
            {
              validator: (_, value: string) => {
                const message = validatePassword(value ?? '');
                return message ? Promise.reject(new Error(message)) : Promise.resolve();
              },
            },
          ]}
        >
          <Input.Password placeholder="请输入密码" autoComplete="current-password" />
        </Form.Item>
        <Button type="primary" htmlType="submit" block loading={submitting} disabled={submitting}>
          登录
        </Button>
      </Form>
      <div style={{ marginTop: 16, textAlign: 'center' }}>
        还没有账号？<Link to="/register">注册</Link>
      </div>
    </div>
  );
}
```

import { Alert, App as AntdApp, Button, Form, Input, Typography } from 'antd';
import { useState } from 'react';
import { Link, useNavigate } from 'react-router';
import { register } from '@/api/auth';
import { ApiError } from '@/api/errors';
import {
  validateConfirmPassword,
  validateEmail,
  validateNickname,
  validatePassword,
} from '@/lib/validators';

// 注册表单
interface RegisterFormValues {
  email: string;
  password: string;
  confirmPassword: string;
  nickname: string;
}

// 注册页
export default function RegisterPage() {
  const [form] = Form.useForm<RegisterFormValues>();
  const navigate = useNavigate();
  const { message } = AntdApp.useApp();
  const [error, setError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState(false);

  // 提交注册：成功后不自动登录
  async function handleFinish(values: RegisterFormValues) {
    setError(null);
    setSubmitting(true);
    try {
      await register(values);
      message.success('注册成功，请登录');
      navigate('/login?registered=1', { replace: true });
    } catch (caught) {
      if (caught instanceof ApiError && (caught.code === 1001 || caught.code === 409)) {
        form.setFields([{ name: 'email', errors: [caught.message] }]);
      } else {
        setError(caught instanceof ApiError ? caught.message : '注册失败，请稍后重试');
      }
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <div style={{ padding: 32, background: '#fff', borderRadius: 6, boxShadow: 'var(--shadow-card)' }}>
      <Typography.Title level={3} style={{ marginTop: 0 }}>
        注册
      </Typography.Title>
      {error && <Alert type="error" showIcon message={error} style={{ marginBottom: 16 }} />}
      <Form form={form} layout="vertical" requiredMark={false} onFinish={handleFinish}>
        <Form.Item
          name="email"
          label="邮箱"
          rules={[
            { required: true, message: '请输入邮箱' },
            {
              validator: (_, value: string) => {
                const message = validateEmail(value ?? '');
                return message ? Promise.reject(new Error(message)) : Promise.resolve();
              },
            },
          ]}
        >
          <Input placeholder="new@example.com" autoComplete="email" />
        </Form.Item>
        <Form.Item
          name="password"
          label="密码"
          rules={[
            { required: true, message: '请输入密码' },
            {
              validator: (_, value: string) => {
                const message = validatePassword(value ?? '');
                return message ? Promise.reject(new Error(message)) : Promise.resolve();
              },
            },
          ]}
        >
          <Input.Password placeholder="8–64 位，至少一个字母和一个数字" autoComplete="new-password" />
        </Form.Item>
        <Form.Item
          name="confirmPassword"
          label="确认密码"
          dependencies={['password']}
          rules={[
            { required: true, message: '请再次输入密码' },
            ({ getFieldValue }) => ({
              validator: (_, value: string) => {
                const message = validateConfirmPassword(getFieldValue('password') ?? '', value ?? '');
                return message ? Promise.reject(new Error(message)) : Promise.resolve();
              },
            }),
          ]}
        >
          <Input.Password placeholder="请再次输入密码" autoComplete="new-password" />
        </Form.Item>
        <Form.Item
          name="nickname"
          label="昵称"
          rules={[
            { required: true, message: '请输入昵称' },
            {
              validator: (_, value: string) => {
                const message = validateNickname(value ?? '');
                return message ? Promise.reject(new Error(message)) : Promise.resolve();
              },
            },
          ]}
        >
          <Input placeholder="最多 30 个字符" maxLength={30} showCount />
        </Form.Item>
        <Button type="primary" htmlType="submit" block loading={submitting} disabled={submitting}>
          注册
        </Button>
      </Form>
      <div style={{ marginTop: 16, textAlign: 'center' }}>
        已有账号？<Link to="/login">去登录</Link>
      </div>
    </div>
  );
}
