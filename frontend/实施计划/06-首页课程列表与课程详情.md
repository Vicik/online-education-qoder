```ts
// src/api/catalog.ts
import { request } from './client';
import type {
  CourseBrief,
  CourseDetail,
  PageResult,
} from './types';

// 分页参数：page 从 1 开始，pageSize 上限 100
export interface PageParams {
  page?: number;
  pageSize?: number;
}

// 已发布课程分页列表（后端不支持搜索/筛选，前端不传额外参数）
export function listCourses(params: PageParams = {}): Promise<PageResult<CourseBrief>> {
  return request<PageResult<CourseBrief>>({
    url: '/courses',
    method: 'GET',
    params,
  });
}

// 首页推荐课程：后端返回 { courses }，此处只暴露数组
export async function getRecommendations(): Promise<CourseBrief[]> {
  // 原始响应数据
  const data = await request<{ courses: CourseBrief[] }>({
    url: '/home/recommendations',
    method: 'GET',
  });
  return data.courses;
}

// 课程详情：未发布或不存在时后端抛 3001
export function getCourseDetail(courseId: number): Promise<CourseDetail> {
  return request<CourseDetail>({
    url: `/courses/${courseId}`,
    method: 'GET',
  });
}

// 我的收藏分页（仅课程）
export function listFavorites(
  page = 1,
  pageSize = 100,
): Promise<PageResult<CourseBrief>> {
  return request<PageResult<CourseBrief>>({
    url: '/user/favorites',
    method: 'GET',
    params: { page, pageSize },
  });
}

// 收藏课程：幂等 POST，契约要求携带 Idempotency-Key
export function addFavorite(
  courseId: number,
  idempotencyKey: string,
): Promise<void> {
  return request<void>({
    url: '/favorites',
    method: 'POST',
    data: { courseId },
    headers: { 'Idempotency-Key': idempotencyKey },
  });
}

// 取消收藏：后端幂等，未收藏也不报错
export function removeFavorite(courseId: number): Promise<void> {
  return request<void>({
    url: `/favorites/${courseId}`,
    method: 'DELETE',
  });
}

// 收藏 id 全集：详情页据此判断收藏态；按 100 条一页扫描，数据量上限由用户收藏数决定
export async function listFavoriteIds(): Promise<Set<number>> {
  // 第一页同时给出总数
  const firstPage = await listFavorites(1, 100);
  // 收藏 id 集合
  const ids = new Set(firstPage.items.map((item) => item.id));
  // 剩余页数
  const totalPages = Math.ceil(firstPage.total / 100);
  for (let page = 2; page <= totalPages; page += 1) {
    // 后续页
    const nextPage = await listFavorites(page, 100);
    nextPage.items.forEach((item) => ids.add(item.id));
  }
  return ids;
}
```

```tsx
// tests/utils/render.tsx
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { App as AntdApp } from 'antd';
import { render } from '@testing-library/react';
import { RouterProvider, createMemoryRouter, type RouteObject } from 'react-router';

// 页面级测试包装：独立 QueryClient（关闭重试）+ 内存路由 + AntD App 上下文
export function renderWithProviders(routes: RouteObject[], initialEntry: string) {
  // 每个用例独立缓存，避免相互污染
  const queryClient = new QueryClient({
    defaultOptions: {
      queries: { retry: false },
      mutations: { retry: false },
    },
  });
  // 内存路由：断言跳转用 router.state.location
  const router = createMemoryRouter(routes, { initialEntries: [initialEntry] });

  render(
    <AntdApp>
      <QueryClientProvider client={queryClient}>
        <RouterProvider router={router} />
      </QueryClientProvider>
    </AntdApp>,
  );

  return { queryClient, router };
}
```

```tsx
// 封面：无地址或加载失败时回退占位块；失败只通知一次父资源重取
function CourseCover({
  url,
  alt,
  onError,
}: {
  url: string | null;
  alt: string;
  onError?: () => void;
}) {
  // 本卡片内是否已回退占位
  const [failed, setFailed] = useState(false);

  if (!url || failed) {
    return <div style={{ height: 150, background: '#f0f2f5' }} aria-label="课程封面占位" />;
  }

  return (
    <img
      src={url}
      alt={alt}
      style={{ width: '100%', height: 150, objectFit: 'cover' }}
      onError={() => {
        setFailed(true);
        onError?.();
      }}
    />
  );
}

// 课程卡片：只展示封面、名称、讲师、价格；不展示课程类型与报名人数
```

```tsx
// src/features/home/HomePage.tsx
import { Typography } from 'antd';
import { Link } from 'react-router';
import { useQuery } from '@tanstack/react-query';
import { queryKeys } from '@/api/query-keys';
import { getRecommendations, listCourses } from '@/api/catalog';
import { CourseGrid, CourseGridSkeleton } from '@/features/catalog/CourseGrid';
import { EmptyState } from '@/components/common/EmptyState';

// 首页最新课程展示数量
const LATEST_SIZE = 8;

// 首页：推荐 + 最新，两块独立请求，互不阻塞
export default function HomePage() {
  // 推荐课程
  const recommendations = useQuery({
    queryKey: queryKeys.homeRecommendations,
    queryFn: getRecommendations,
  });
  // 最新课程（独立键，避免与课程列表页共享分页缓存）
  const latest = useQuery({
    queryKey: queryKeys.homeLatest,
    queryFn: () => listCourses({ page: 1, pageSize: LATEST_SIZE }),
  });

  return (
    <div>
      <Typography.Title level={2} style={{ marginTop: 0 }}>
        知序课堂
      </Typography.Title>
      <Typography.Paragraph type="secondary">结构化课程，按序学习。</Typography.Paragraph>

      <Typography.Title level={4}>推荐课程</Typography.Title>
      {recommendations.isLoading ? (
        <CourseGridSkeleton />
      ) : recommendations.data && recommendations.data.length > 0 ? (
        <CourseGrid courses={recommendations.data} onCoverError={() => recommendations.refetch()} />
      ) : (
        <EmptyState description="暂无推荐课程" />
      )}

      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginTop: 32 }}>
        <Typography.Title level={4} style={{ margin: 0 }}>
          最新课程
        </Typography.Title>
        <Link to="/courses">查看全部</Link>
      </div>
      {latest.isLoading ? (
        <CourseGridSkeleton />
      ) : latest.data && latest.data.items.length > 0 ? (
        <CourseGrid courses={latest.data.items} onCoverError={() => latest.refetch()} />
      ) : (
        <EmptyState description="暂无课程" />
      )}
    </div>
  );
}
```

```tsx
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { addFavorite, listFavoriteIds, removeFavorite } from '@/api/catalog';
import { queryKeys } from '@/api/query-keys';
import { useIdempotentMutation } from '@/hooks/useIdempotentMutation';
import { useSessionStore } from '@/stores/session';

// 收藏失效：一次覆盖 id 集与各页列表
function invalidateFavorites(client: ReturnType<typeof useQueryClient>): Promise<void> {
  return client.invalidateQueries({ queryKey: queryKeys.favoritesRoot });
}

// 收藏 id 全集：仅登录用户请求
export function useFavoriteIds() {
  // 当前会话用户
  const user = useSessionStore((state) => state.user);
  return useQuery({
    queryKey: queryKeys.favoriteIds,
    queryFn: listFavoriteIds,
    enabled: Boolean(user),
  });
}

// 收藏切换：POST 走幂等，DELETE 幂等；成功后失效收藏缓存
export function useFavoriteToggle(courseId: number) {
  // QueryClient：驱动缓存失效
  const queryClient = useQueryClient();
  // 收藏 id 集
  const { data: ids } = useFavoriteIds();
  // 当前课程是否已收藏（未拿到 id 集时按未收藏展示）
  const favorited = ids?.has(courseId) ?? false;

  // 收藏突变
  const addMutation = useIdempotentMutation(
    (id: number, idempotencyKey: string) => addFavorite(id, idempotencyKey),
    { onSuccess: () => invalidateFavorites(queryClient) },
  );
  // 取消收藏突变
  const removeMutation = useMutation({
    mutationFn: removeFavorite,
    onSuccess: () => invalidateFavorites(queryClient),
  });

  // 请求进行中：禁用按钮防重复提交
  const pending = addMutation.isPending || removeMutation.isPending;

  // 切换收藏态
  async function toggle(): Promise<void> {
    if (favorited) {
      await removeMutation.mutateAsync(courseId);
    } else {
      await addMutation.mutateAsync(courseId);
    }
  }

  return { favorited, pending, toggle };
}
```

```tsx
import { Button, Pagination } from 'antd';
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { useEffect, useState } from 'react';
import { listFavorites, removeFavorite } from '@/api/catalog';
import { queryKeys } from '@/api/query-keys';
import { ConfirmAction } from '@/components/common/ConfirmAction';
import { EmptyState } from '@/components/common/EmptyState';
import { PageHeader } from '@/components/common/PageHeader';
import { usePagination } from '@/hooks/usePagination';
import { CourseGridSkeleton } from '@/features/catalog/CourseGrid';
import { CourseCard } from '@/features/catalog/CourseCard';

// 收藏页每页数量
const PAGE_SIZE = 12;

// 我的收藏：仅课程，取消收藏需二次确认
export default function FavoritesPage() {
  // QueryClient：删除后失效收藏缓存
  const queryClient = useQueryClient();
  // 收藏总数（首屏 0，响应后回填）
  const [total, setTotal] = useState(0);
  // 分页状态
  const { page, pageSize, onChange } = usePagination(total, PAGE_SIZE);
  // 当前页收藏
  const query = useQuery({
    queryKey: queryKeys.favorites(page),
    queryFn: () => listFavorites(page, PAGE_SIZE),
    placeholderData: (previous) => previous,
  });
  // 汇总取消收藏
  const remove = useMutation({
    mutationFn: removeFavorite,
    onSuccess: () => queryClient.invalidateQueries({ queryKey: queryKeys.favoritesRoot }),
  });

  // 响应总数同步给分页组件
  useEffect(() => {
    setTotal(query.data?.total ?? 0);
  }, [query.data?.total]);

  return (
    <div>
      <PageHeader title="我的收藏" />
      {query.isLoading ? (
        <CourseGridSkeleton count={3} />
      ) : query.data && query.data.items.length > 0 ? (
        <>
          <div
            style={{
              display: 'grid',
              gridTemplateColumns: 'repeat(auto-fill, minmax(240px, 1fr))',
              gap: 16,
            }}
          >
            {query.data.items.map((course) => (
              <CourseCard
                key={course.id}
                course={course}
                onCoverError={() => query.refetch()}
                footer={
                  <ConfirmAction
                    title="取消收藏该课程？"
                    danger
                    onConfirm={() => remove.mutateAsync(course.id)}
                    disabled={remove.isPending}
                  >
                    <Button danger block style={{ marginTop: 8 }}>
                      取消收藏
                    </Button>
                  </ConfirmAction>
                }
              />
            ))}
          </div>
          <div style={{ display: 'flex', justifyContent: 'center', marginTop: 24 }}>
            <Pagination
              current={page}
              pageSize={pageSize}
              total={total}
              showSizeChanger={false}
              onChange={onChange}
            />
          </div>
        </>
      ) : (
        <EmptyState description="暂无收藏" action={<a href="/courses">去逛逛</a>} />
      )}
    </div>
  );
}
```
