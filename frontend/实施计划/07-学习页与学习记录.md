```ts
import { request } from './client';
import type { ChapterContent, PageResult, StudyRecord } from './types';

// 学习进度：上报访问后的权威结果
export interface LearningProgress {
  progressPercent: number; // 已访问章节数 / 总章节数
  completed: boolean; // 全部章节至少访问一次
}

// 章节正文：需已购买；未购 403（3002）
export function getChapterContent(
  courseId: number,
  chapterId: number,
): Promise<ChapterContent> {
  return request<ChapterContent>({
    url: `/courses/${courseId}/chapters/${chapterId}/content`,
    method: 'GET',
  });
}

// 记录章节访问：后端 upsert 幂等，无需 Idempotency-Key
export function recordChapterVisit(
  courseId: number,
  chapterId: number,
): Promise<LearningProgress> {
  return request<LearningProgress>({
    url: `/learning/${courseId}/chapters/${chapterId}/visit`,
    method: 'POST',
  });
}

// 学习记录分页：按最近学习时间倒序
export function listStudyRecords(
  page = 1,
  pageSize = 20,
): Promise<PageResult<StudyRecord>> {
  return request<PageResult<StudyRecord>>({
    url: '/user/study-records',
    method: 'GET',
    params: { page, pageSize },
  });
}
```

```tsx
// src/features/learning/ChapterNav.tsx
import { List, Tag } from 'antd';
import type { ChapterBrief } from '@/api/types';

// 章节导航：点击切换，已访问章节标「已学」
export function ChapterNav({
  chapters,
  activeChapterId,
  visitedChapterIds,
  onSelect,
}: {
  chapters: ChapterBrief[];
  activeChapterId: number | null;
  visitedChapterIds: Set<number>;
  onSelect: (chapterId: number) => void;
}) {
  return (
    <List
      dataSource={chapters}
      renderItem={(chapter, index) => {
        // 后端如返回 visited 则以其为准，否则用本地已访问集合
        const visited = chapter.visited === true || visitedChapterIds.has(chapter.id);
        return (
          <List.Item
            style={{
              cursor: 'pointer',
              paddingInline: 12,
              background: chapter.id === activeChapterId ? '#e6f4ff' : undefined,
            }}
            onClick={() => onSelect(chapter.id)}
          >
            <div style={{ display: 'flex', justifyContent: 'space-between', width: '100%' }}>
              <span>
                {index + 1}. {chapter.title}
              </span>
              {visited && <Tag color="green">已学</Tag>}
            </div>
          </List.Item>
        );
      }}
    />
  );
}
```

```tsx
import { EmptyState } from '@/components/common/EmptyState';
import { RichTextViewer } from '@/components/common/RichTextViewer';
import type { ChapterContent } from '@/api/types';

// 章节正文：图文用净化后的富文本；视频/音频用签名地址
export function ChapterViewer({ content }: { content: ChapterContent }) {
  if (!content.content) {
    return <EmptyState description="章节内容暂未上传" />;
  }

  if (content.contentType === 'rich') {
    return <RichTextViewer html={content.content} />;
  }

  if (content.contentType === 'video') {
    return (
      <video
        key={content.content}
        controls
        style={{ width: '100%', maxHeight: 480 }}
        src={content.content}
      />
    );
  }

  return <audio controls style={{ width: '100%' }} src={content.content} />;
}
```

```tsx
// src/features/learning/LearningPage.tsx
import { Button, Card, Col, Progress, Row, Space, Tag, Typography } from 'antd';
import { useMutation, useQuery } from '@tanstack/react-query';
import { useEffect, useMemo, useRef, useState } from 'react';
import { useNavigate, useParams, useSearchParams } from 'react-router';
import { getCourseDetail } from '@/api/catalog';
import { ApiError, getErrorAction } from '@/api/errors';
import { getChapterContent, listStudyRecords, recordChapterVisit, type LearningProgress } from '@/api/learning';
import { queryKeys } from '@/api/query-keys';
import type { ChapterBrief } from '@/api/types';
import { EmptyState } from '@/components/common/EmptyState';
import { useQueryClient } from '@tanstack/react-query';
import { ChapterNav } from './ChapterNav';
import { ChapterViewer } from './ChapterViewer';

// 初始章节：优先 URL 指定，其次学习记录中的最近章节，最后第一章
export function resolveInitialChapterId(
  chapters: ChapterBrief[],
  requestedChapterId: number | null,
  lastChapterId: number | null | undefined,
): number | null {
  // URL 指定且属于本课程
  if (requestedChapterId !== null && chapters.some((chapter) => chapter.id === requestedChapterId)) {
    return requestedChapterId;
  }
  // 学习记录的最近章节
  if (lastChapterId != null && chapters.some((chapter) => chapter.id === lastChapterId)) {
    return lastChapterId;
  }
  // 兜底第一章
  return chapters[0]?.id ?? null;
}

// 学习页：左侧章节导航 + 右侧正文，打开即上报访问
export default function LearningPage() {
  // 路由课程 id
  const { courseId: rawCourseId } = useParams();
  // 解析后的课程 id
  const courseId = Number(rawCourseId);
  // URL 中的章节 id
  const [searchParams, setSearchParams] = useSearchParams();
  // 请求的章节 id（非法时为 null）
  const requestedChapterId = Number(searchParams.get('chapterId')) || null;
  // 跳转
  const navigate = useNavigate();
  // QueryClient：访问成功后失效学习记录
  const queryClient = useQueryClient();
  // 已访问章节（本地即时标记）
  const [visitedChapterIds, setVisitedChapterIds] = useState<Set<number>>(new Set());
  // 最近一次上报返回的进度
  const [progress, setProgress] = useState<LearningProgress | null>(null);
  // 已上报的章节 id，防止同一章节重复上报
  const reportedChapterRef = useRef<number | null>(null);

  // 课程详情（章节名与标题）
  const courseQuery = useQuery({
    queryKey: queryKeys.course(courseId),
    queryFn: () => getCourseDetail(courseId),
    enabled: Number.isInteger(courseId) && courseId > 0,
  });
  // 学习记录：仅用于取最近章节
  const recordsQuery = useQuery({
    queryKey: queryKeys.studyRecords(1),
    queryFn: () => listStudyRecords(1, 100),
    enabled: Number.isInteger(courseId) && courseId > 0,
  });

  // 最近学习章节
  const lastChapterId = recordsQuery.data?.items.find((item) => item.courseId === courseId)?.lastChapterId;

  // 无 URL 指定章节时，等学习记录就绪再定初始章节，避免先打开第一章再跳走导致重复上报
  const recordsReady = requestedChapterId !== null || recordsQuery.isFetched;

  // 章节列表（稳定引用）
  const chapters = useMemo(() => courseQuery.data?.chapters ?? [], [courseQuery.data]);

  // 当前章节 id
  const activeChapterId = resolveInitialChapterId(chapters, requestedChapterId, lastChapterId);

  // 当前章节正文
  const contentQuery = useQuery({
    queryKey: queryKeys.courseContent(courseId, activeChapterId ?? 0),
    queryFn: () => getChapterContent(courseId, activeChapterId as number),
    enabled: activeChapterId !== null && recordsReady && Number.isInteger(courseId) && courseId > 0,
    retry: false,
  });

  // 上报访问
  const visitMutation = useMutation({
    mutationFn: (chapterId: number) => recordChapterVisit(courseId, chapterId),
    onSuccess: (result, chapterId) => {
      setProgress(result);
      setVisitedChapterIds((previous) => new Set(previous).add(chapterId));
      queryClient.invalidateQueries({ queryKey: queryKeys.studyRecordsRoot });
    },
  });

  // 正文加载成功后上报一次（同一章节只报一次）
  useEffect(() => {
    if (!contentQuery.data || activeChapterId === null) return;
    if (reportedChapterRef.current === activeChapterId) return;
    reportedChapterRef.current = activeChapterId;
    visitMutation.mutate(activeChapterId);
  }, [contentQuery.data, activeChapterId]);

  // 切换章节：写入 URL，刷新后保持
  function selectChapter(chapterId: number): void {
    setSearchParams({ chapterId: String(chapterId) });
  }

  // 未购买：正文接口返回 3002
  if (contentQuery.error instanceof ApiError && getErrorAction(contentQuery.error) === 'not-purchased') {
    return (
      <EmptyState
        description="购买后可学习"
        action={
          <Space>
            <Button type="primary" onClick={() => navigate(`/checkout/${courseId}`)}>
              去购买
            </Button>
            <Button onClick={() => navigate(`/courses/${courseId}`)}>查看课程详情</Button>
          </Space>
        }
      />
    );
  }

  if (courseQuery.isLoading) {
    return <Card loading />;
  }

  if (courseQuery.error || !courseQuery.data) {
    return (
      <EmptyState
        description="课程不存在或已下架"
        action={<Button onClick={() => navigate('/courses')}>返回课程列表</Button>}
      />
    );
  }

  // 进度百分比：优先上报结果，其次学习记录
  const progressPercent = progress?.progressPercent
    ?? recordsQuery.data?.items.find((item) => item.courseId === courseId)?.progressPercent
    ?? 0;
  // 是否已学完
  const completed = progress?.completed
    ?? recordsQuery.data?.items.find((item) => item.courseId === courseId)?.completed
    ?? false;
  // 当前章节索引
  const activeIndex = chapters.findIndex((chapter) => chapter.id === activeChapterId);

  return (
    <div>
      <Card style={{ marginBottom: 16 }}>
        <Space direction="vertical" style={{ width: '100%' }} size={4}>
          <Typography.Title level={4} style={{ margin: 0 }}>
            {courseQuery.data.name}
          </Typography.Title>
          <Space>
            <Progress percent={progressPercent} style={{ width: 240 }} />
            {completed && <Tag color="green">已学完</Tag>}
          </Space>
        </Space>
      </Card>

      <Row gutter={16}>
        <Col flex="240px">
          <Card title="章节目录" styles={{ body: { padding: 0 } }}>
            <ChapterNav
              chapters={chapters}
              activeChapterId={activeChapterId}
              visitedChapterIds={visitedChapterIds}
              onSelect={selectChapter}
            />
          </Card>
        </Col>
        <Col flex="auto">
          <Card
            title={chapters.find((chapter) => chapter.id === activeChapterId)?.title}
            extra={
              <Space>
                <Button
                  disabled={activeIndex <= 0}
                  onClick={() => selectChapter(chapters[activeIndex - 1].id)}
                >
                  上一章
                </Button>
                <Button
                  disabled={activeIndex < 0 || activeIndex >= chapters.length - 1}
                  onClick={() => selectChapter(chapters[activeIndex + 1].id)}
                >
                  下一章
                </Button>
              </Space>
            }
          >
            {contentQuery.isLoading ? (
              <Card loading />
            ) : contentQuery.data ? (
              <ChapterViewer content={contentQuery.data} />
            ) : contentQuery.error ? (
              <EmptyState description={contentQuery.error.message} />
            ) : (
              <EmptyState description="暂无章节" />
            )}
          </Card>
        </Col>
      </Row>
    </div>
  );
}
```

```tsx
import { Button, Progress, Tag } from 'antd';
import { useQuery } from '@tanstack/react-query';
import { useEffect, useState } from 'react';
import { useNavigate } from 'react-router';
import { listStudyRecords } from '@/api/learning';
import { queryKeys } from '@/api/query-keys';
import type { StudyRecord } from '@/api/types';
import { AppTable } from '@/components/common/AppTable';
import { PageHeader } from '@/components/common/PageHeader';
import { formatShanghai } from '@/lib/datetime';
import { usePagination } from '@/hooks/usePagination';

// 学习记录：按课程聚合的进度与最近章节
export default function StudyRecordsPage() {
  // 跳转
  const navigate = useNavigate();
  // 记录总数（首屏 0，响应后回填）
  const [total, setTotal] = useState(0);
  // 分页状态
  const { page, pageSize, onChange } = usePagination(total, 20);
  // 学习记录
  const query = useQuery({
    queryKey: queryKeys.studyRecords(page),
    queryFn: () => listStudyRecords(page, pageSize),
  });

  // 响应总数同步给分页组件
  useEffect(() => {
    setTotal(query.data?.total ?? 0);
  }, [query.data?.total]);

  return (
    <div>
      <PageHeader title="学习记录" />
      <AppTable<StudyRecord>
        rowKey="courseId"
        dataSource={query.data?.items ?? []}
        loading={query.isLoading}
        error={query.error}
        onRetry={() => query.refetch()}
        total={total}
        page={page}
        pageSize={pageSize}
        onPageChange={onChange}
        emptyText="暂无学习记录"
        columns={[
          { title: '课程', dataIndex: 'courseName' },
          {
            title: '进度',
            dataIndex: 'progressPercent',
            width: 220,
            render: (value: number) => <Progress percent={value} size="small" />,
          },
          {
            title: '状态',
            dataIndex: 'completed',
            width: 100,
            render: (completed: boolean) =>
              completed ? <Tag color="green">已学完</Tag> : <Tag>学习中</Tag>,
          },
          {
            title: '最近章节',
            dataIndex: 'lastChapterTitle',
            render: (title: string | null) => title ?? '—',
          },
          {
            title: '最后学习时间',
            dataIndex: 'lastVisitedAt',
            render: (value: string) => formatShanghai(value),
          },
          {
            title: '操作',
            width: 120,
            render: (_: unknown, record: StudyRecord) => (
              <Button
                type="link"
                onClick={() =>
                  navigate(
                    `/learning/${record.courseId}${record.lastChapterId ? `?chapterId=${record.lastChapterId}` : ''}`,
                  )
                }
              >
                继续学习
              </Button>
            ),
          },
        ]}
      />
    </div>
  );
}
```
