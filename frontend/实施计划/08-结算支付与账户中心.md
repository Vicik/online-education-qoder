  courseId: number,
  query: CheckoutQuery = {},
): Promise<CheckoutPreviewData> {
  return request<CheckoutPreviewData>({
    url: `/checkout/${courseId}/preview`,
    method: 'GET',
    params: query,
  });

export function createOrder(
  payload: OrderCreateRequest,
  idempotencyKey: string,
): Promise<Order> {
  return request<Order>({
    url: '/orders',
    method: 'POST',
    data: payload,
    headers: { 'Idempotency-Key': idempotencyKey },
  });
}

// 模拟支付：后端幂等，已支付返回首次结果
export function payOrder(orderId: number): Promise<Order> {
  return request<Order>({
    url: `/orders/${orderId}/pay`,
    method: 'POST',
  });
}

  return request<Order>({
    url: `/orders/${orderId}`,

        value: coupon.id,
        label: `${coupon.templateName}（立减 ${formatCents(coupon.discountCents)}）`,
      })) ?? [];

  // 提交：创建（或取既有）订单后支付
  async function handleSubmit(): Promise<void> {
    setSubmitError(null);
    try {
      // 待支付订单
      let order: Order;
      if (existingOrderId !== null) {
        order = orderQuery.data as Order;
      } else {
        order = await createMutation.mutateAsync({
          courseId,
          activityType: previewQuery.data?.activityType ?? 'normal',
          activityId: previewQuery.data?.activityId ?? null,
          couponClaimId: couponId ?? null,
        });
      }

      await payMutation.mutateAsync(order.id);
      queryClient.invalidateQueries({ queryKey: queryKeys.myOrders(1) });
      queryClient.invalidateQueries({ queryKey: queryKeys.overview });
      queryClient.invalidateQueries({ queryKey: queryKeys.myCoupons('unused', 1) });
      navigate(`/order/${order.id}/result`, { replace: true });
    } catch (error) {
      if (!(error instanceof ApiError)) {
        setSubmitError('支付失败，请稍后重试');
        return;
      }
      if (error.code === 4003) {
        setDuplicatePurchase(true);
        message.error('您已购买该课程');
        return;
      }
      if (error.code === 4002) {
        message.warning('优惠券不可用，已刷新价格');
        setCouponId(undefined);
        await previewQuery.refetch();
        return;
      }
      if (error.code === 4001 || error.code === 5003) {
        setSubmitError(error.message);
        return;
      }
      if (error.code === 4004) {
        message.info('请勿重复提交');
        return;
      }
      setSubmitError(error.message);
    }
  }

  if (previewQuery.error && !existingOrderId) {
    return <EmptyState description={previewQuery.error.message} />;
  }

  if (!amountSource) {
    return <Card loading />;
  }

  if (duplicatePurchase) {
    return (
      <EmptyState
        description="您已购买该课程"
        action={
          <Link to={`/learning/${courseId}`}>
            <Button type="primary">去学习</Button>
          </Link>
        }
      />
    );
  }

  if (existingOrderId !== null && orderQuery.data && orderQuery.data.status !== 'pending') {
    return (
      <EmptyState
        description="该订单已处理"
        action={<Link to={`/order/${existingOrderId}/result`}>查看支付结果</Link>}
      />
    );
  }

  return (
    <div>
      <PageHeader title="确认订单" />
      <Card style={{ maxWidth: 640 }}>
        <Typography.Title level={5}>{orderQuery.data?.courseName ?? '课程订单'}</Typography.Title>
        <PriceBreakdown preview={amountSource} />

        {existingOrderId === null && (
          <div style={{ marginTop: 16 }}>
            <Typography.Text>优惠券</Typography.Text>
            <Select
              allowClear
              style={{ width: '100%', marginTop: 8 }}
              placeholder="不使用优惠券"
              value={couponId}
              onChange={(value) => setCouponId(value)}
              options={couponOptions}
            />
          </div>
        )}

        {submitError && <Alert type="error" showIcon message={submitError} style={{ marginTop: 16 }} />}

        <div style={{ marginTop: 24, textAlign: 'right' }}>
          <ConfirmAction
            title={`${submitLabel} ${formatCents(paidCents)}？`}
            content="支付为模拟支付，确认后将立即完成订单。"
            okText="确定"

              size="large"

              loading={createMutation.isPending || payMutation.isPending}

      message.success('保存成功');

      email: query.data.email,
      nickname: query.data.nickname,
      gender: query.data.gender,
      avatar: query.data.avatarKey
        ? { key: query.data.avatarKey, url: query.data.avatarUrl }
        : null,
    });
  }, [query.data, form]);

  // 提交：头像写回 key，昵称同步会话
  async function handleFinish(values: ProfileFormValues): Promise<void> {
    await mutation.mutateAsync({
      nickname: values.nickname,
      avatarUrl: values.avatar?.key ?? null,
      gender: values.gender,
    });
    if (user) {
      setUser({
        ...user,
        nickname: values.nickname,
        avatarKey: values.avatar?.key ?? null,
        avatarUrl: values.avatar?.url ?? null,
      });
    }
  }

  if (query.isLoading) return <Card loading />;

  return (
    <div>
      <PageHeader title="修改资料" />
      <Card style={{ maxWidth: 640 }}>
        <Form form={form} layout="vertical" onFinish={handleFinish}>
          <Form.Item name="email" label="邮箱">

            保存
          </Button>
        </Form>

  );
}

// src/features/user-center/PasswordPage.tsx
import { Alert, App as AntdApp, Button, Card, Form, Input } from 'antd';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { useState } from 'react';
import { useNavigate } from 'react-router';
import { changePassword } from '@/api/auth';
import { validatePassword } from '@/lib/validators';
import { useSessionStore } from '@/stores/session';
import { PageHeader } from '@/components/common/PageHeader';

// 改密表单值
interface PasswordFormValues {
  currentPassword: string;
  newPassword: string;
  confirmPassword: string;
}

// 修改密码：成功后后端吊销会话，前端清空状态并回登录页
export default function PasswordPage() {
  // 表单实例
  const [form] = Form.useForm<PasswordFormValues>();
  // 跳转
  const navigate = useNavigate();
  // QueryClient
  const queryClient = useQueryClient();
  // 清空会话
  const clearSession = useSessionStore((state) => state.clear);
  // AntD message
  const { message } = AntdApp.useApp();
  // 业务错误
  const [error, setError] = useState<string | null>(null);

  // 改密请求
  const mutation = useMutation({
    mutationFn: changePassword,
    onSuccess: () => {
      clearSession();
      queryClient.clear();
      message.success('密码已修改，请重新登录');
      navigate('/login?changed=1', { replace: true });
    },
    onError: (caught: Error) => {
      setError(caught.message);
    },
  });

  return (
    <div>
      <PageHeader title="修改密码" />
      <Card style={{ maxWidth: 520 }}>
        {error && <Alert type="error" showIcon message={error} style={{ marginBottom: 16 }} />}
        <Form
          form={form}
          layout="vertical"
          onFinish={(values: PasswordFormValues) => mutation.mutate(values)}
        >
          <Form.Item
            name="currentPassword"
            label="当前密码"
            rules={[{ required: true, message: '请输入当前密码' }]}
          >
            <Input.Password autoComplete="current-password" />
          </Form.Item>
          <Form.Item
            name="newPassword"
            label="新密码"
            rules={[
              { required: true, message: '请输入新密码' },
              {
                validator: (_, value: string) => {
                  const message = validatePassword(value ?? '');
                  return message ? Promise.reject(new Error(message)) : Promise.resolve();
                },
              },
            ]}
          >
            <Input.Password autoComplete="new-password" />
          </Form.Item>
          <Form.Item
            name="confirmPassword"
            label="确认新密码"
            dependencies={['newPassword']}
            rules={[
              { required: true, message: '请再次输入新密码' },
              ({ getFieldValue }) => ({
                validator: (_, value: string) =>
                  getFieldValue('newPassword') === value
                    ? Promise.resolve()
                    : Promise.reject(new Error('两次输入的密码不一致')),
              }),
            ]}
          >
            <Input.Password autoComplete="new-password" />
          </Form.Item>
          <Button type="primary" htmlType="submit" loading={mutation.isPending}>
            修改密码
          </Button>
        </Form>
      </Card>
    </div>
  );
}
