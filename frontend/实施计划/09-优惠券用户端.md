**后端行为约定（已确认）**：
- `GET /courses/:id/coupons` 公开，返回启用、有效期内、适用该课程且未发完的模板。
- `POST /coupons/:templateId/claim` 必带 `Idempotency-Key`；成功 201；同一模板每人限领一张，重复领取为 `409`（`code=1001`）；停用/过期/领完为 `422`（`code=4002`）。

```ts
}
```

```tsx
// src/features/coupon/ClaimCouponButton.tsx
import { useState } from 'react';
import { useNavigate } from 'react-router';
import { Button, message } from 'antd';
import { useQueryClient } from '@tanstack/react-query';
import { claimCoupon } from '@/api/promotion';
import type { CouponClaim, CouponTemplateBrief } from '@/api/types';
import { ConfirmAction } from '@/components/common/ConfirmAction';
import { useIdempotentMutation } from '@/hooks/useIdempotentMutation';
import { usePermission } from '@/hooks/usePermission';
import { queryKeys } from '@/api/query-keys';

// 课程详情领券按钮：游客登录回跳、登录幂等领取、限领/不可领取提示
export function ClaimCouponButton(props: { coupon: CouponTemplateBrief; courseId: number }) {
  // 路由跳转（登录回跳）
  const navigate = useNavigate();
  // 查询缓存（领取成功后失效我的券）
  const queryClient = useQueryClient();
  // 当前用户（仅用于展示判断，权限边界在路由层）
  const { user } = usePermission();
  // 已领取标记：成功或后端 409 后置位
  const [claimed, setClaimed] = useState(false);

  // 幂等领取：同一意图复用键，成功后自动重置
  const mutation = useIdempotentMutation<CouponClaim, void>(
    (_vars, key) => claimCoupon(props.coupon.id, key),
    {
      onSuccess: () => {
        setClaimed(true);
        message.success('领取成功');
        // 我的券所有页签失效
        void queryClient.invalidateQueries({ queryKey: queryKeys.myCouponsRoot });
      },
      onError: (error) => {
        // 409：该模板已领取过（每人限领一张）
        if (error.status === 409) {
          setClaimed(true);
          message.info('该优惠券每人限领一张');
          return;
        }
        // 4002：停用/过期/已领完
        message.error(error.message);
      },
    },
  );

  // 游客：跳登录并回跳当前课程详情
  if (!user) {
    return (
      <Button
        type="link"
        onClick={() =>
          navigate(`/login?redirect=${encodeURIComponent(`/courses/${props.courseId}`)}`)
        }
      >
        登录领取
      </Button>
    );
  }

  // 已领取：禁用展示
  if (claimed) return <Button size="small" disabled>已领取</Button>;

  return (
    <ConfirmAction
      title="领取优惠券"
      content={`确认领取「${props.coupon.name}」？`}
      onConfirm={async () => {
        try {
          await mutation.mutateAsync();
        } catch {
          // 具体错误已由 onError 提示，这里只负责关闭确认框
        }
      }}
    >
      <Button size="small" type="primary" loading={mutation.isPending}>领取</Button>
    </ConfirmAction>
  );
}
```

```tsx
// src/features/coupon/MyCouponsPage.tsx
import { useEffect, useState } from 'react';
import { List, Tabs } from 'antd';
import { useQuery } from '@tanstack/react-query';
import { listMyCoupons, type CouponStatus } from '@/api/promotion';
import { queryKeys } from '@/api/query-keys';
import { EmptyState } from '@/components/common/EmptyState';
import { PageHeader } from '@/components/common/PageHeader';
import { usePagination } from '@/hooks/usePagination';
import { CouponCard } from './CouponCard';

// 四个状态页签：与设计框架 §8.3 一致，restored 不展示
const TABS: Array<{ key: CouponStatus; label: string }> = [
  { key: 'unused', label: '未使用' },
  { key: 'used', label: '已使用' },
  { key: 'expired', label: '已过期' },
  { key: 'invalid', label: '已失效' },
];

// 我的优惠券：四状态页签 + 分页；每页记录数
const PAGE_SIZE = 10;

// 我的优惠券页面
export default function MyCouponsPage() {
  // 当前状态页签
  const [status, setStatus] = useState<CouponStatus>('unused');
  // 当前状态下的总数（分页入参）
  const [total, setTotal] = useState(0);
  // 页码与页大小
  const { page, pageSize, onChange, reset } = usePagination(total, PAGE_SIZE);

  // 我的券分页查询
  const query = useQuery({
    queryKey: queryKeys.myCoupons(status, page),
    queryFn: () => listMyCoupons(status, page, pageSize),
  });

  // 同步后端总数到分页状态
  useEffect(() => {
    if (query.data) setTotal(query.data.total);
  }, [query.data]);

  // 切换页签：回到第一页并清空旧总数
  function handleTabChange(key: string) {
    setStatus(key as CouponStatus);
    setTotal(0);
    reset();
  }

  // 当前页记录
  const items = query.data?.items ?? [];

  return (
    <div>
      <PageHeader title="我的优惠券" />
      <Tabs
        activeKey={status}
        items={TABS.map((tab) => ({ key: tab.key, label: tab.label }))}
        onChange={handleTabChange}
      />
      {items.length === 0 && !query.isLoading ? (
        <EmptyState description="暂无优惠券" />
      ) : (
        <List
          loading={query.isLoading}
          dataSource={items}
          pagination={{
            current: page,
            pageSize,
            total,
            onChange,
          }}
          renderItem={(item) => <CouponCard coupon={item} />}
        />
      )}
    </div>
```
