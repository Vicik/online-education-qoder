**后端行为约定（已确认）**：
- `GET /groupons` 公开，仅返回 `status=ongoing`，按开始时间正序分页。
- `GET /groupons/:id` 公开，返回活动 + 仅 `pending` 的团组；不存在为 404（`code=1002`）。
- `POST /groupons/:id/groups` 与 `POST /groupons/:id/groups/:gid/join` 必带 `Idempotency-Key`，成功 201 返回 `{ order, group }`，订单为 `pending`，需再走支付。
- 同一活动一人一次：已参与为 `409 + code=5001`；活动未开始/已结束为 `422 + code=5002`；团满为 `422 + code=5001`。
- 团组 `memberCount` 只统计支付成功成员；订单 `activityId` 为团组 id（发起新团时由后端回写）。

```tsx
// src/features/groupon/GrouponListPage.tsx
import { useEffect, useState } from 'react';
import { Col, Pagination, Row } from 'antd';
import { useQuery } from '@tanstack/react-query';
import { listGroupons } from '@/api/promotion';
import { queryKeys } from '@/api/query-keys';
import { EmptyState } from '@/components/common/EmptyState';
import { PageHeader } from '@/components/common/PageHeader';
import { usePagination } from '@/hooks/usePagination';
import { GrouponCard } from './GrouponCard';

// 拼团列表每页数量
const PAGE_SIZE = 12;

// 拼团列表：进行中活动 + 分页 + 服务端倒计时
export default function GrouponListPage() {
  // 当前总数（分页入参）
  const [total, setTotal] = useState(0);
  // 页码与页大小
  const { page, pageSize, onChange } = usePagination(total, PAGE_SIZE);

  // 进行中活动分页查询
  const query = useQuery({
    queryKey: queryKeys.groupons(page),
    queryFn: () => listGroupons(page, pageSize),
  });

  // 同步后端总数
  useEffect(() => {
    if (query.data) setTotal(query.data.total);
  }, [query.data]);

  // 倒计时到期：重取一次列表，剔除已结束活动
  const handleExpire = () => {
    void query.refetch();
  };

  // 当前页活动
  const items = query.data?.items ?? [];

  return (
    <div>
      <PageHeader title="拼团活动" />
      {items.length === 0 && !query.isLoading ? (
        <EmptyState description="暂无进行中的拼团活动" />
      ) : (
        <Row gutter={[16, 16]}>
          {items.map((item) => (
            <Col key={item.id} xs={24} md={12} lg={8}>
              <GrouponCard activity={item} onExpire={handleExpire} />
            </Col>
          ))}
        </Row>
      )}
      <Pagination
        current={page}
        pageSize={pageSize}
        total={total}
        onChange={onChange}
        style={{ marginTop: 16, textAlign: 'right' }}
      />
    </div>
  );
}
```

```tsx
export default function GrouponDetailPage() {
  // 路由参数
  const params = useParams();
  // 活动 id
  const activityId = Number(params.id);
  // 路由跳转
  const navigate = useNavigate();
  // 查询缓存
  const queryClient = useQueryClient();
  // 当前用户
  const { user } = usePermission();

  // 活动详情（含待成团团组）
  const query = useQuery({
    queryKey: queryKeys.groupon(activityId),
    queryFn: () => getGroupon(activityId),
    enabled: Number.isInteger(activityId) && activityId > 0,
  });

  // 失败提示并按错误码刷新详情
  const handleError = (error: ApiError) => {
    // 已参与（409 + 5001）
    if (error.status === 409 && error.code === 5001) {
      message.warning('您已参与过该拼团活动');
    } else if (error.code === 5002) {
      // 活动未开始或已结束
      message.warning('拼团活动未开始或已结束');
    } else if (error.code === 5001) {
      // 团满（422 + 5001）
      message.warning('该团已满或已结束');
    } else if (error.code === 4004) {
      // 幂等冲突：上一次请求仍在处理
      message.warning('请求正在处理中，请稍后重试');
    } else {
      message.error(error.message);
    }
    void query.refetch();
  };

  // 开团/参团共用幂等提交；groupId 为 null 表示开团
  const mutation = useIdempotentMutation<
    GrouponOrderResult,
    { groupId: number | null }
  >(
    (vars, key) =>
      vars.groupId === null
        ? createGrouponGroup(activityId, key)
        : joinGrouponGroup(activityId, vars.groupId, key),
    {
      onSuccess: ({ order }) => {
        // 团组与活动数据已变化，失效详情与列表
        void queryClient.invalidateQueries({ queryKey: queryKeys.groupon(activityId) });
        void queryClient.invalidateQueries({ queryKey: ['groupons'] });
        // 进入既有订单支付模式
        navigate(`/checkout/${order.courseId}?orderId=${order.id}`);
      },
      onError: handleError,
    },
  );

  // 游客：跳登录并回跳详情
  const gotoLogin = () => {
    navigate(`/login?redirect=${encodeURIComponent(`/groupons/${activityId}`)}`);
  };

  if (query.isLoading) return <Card loading />;
  if (query.error || !query.data) {
    return <EmptyState description="拼团活动不存在" action={<Button onClick={() => navigate('/groupons')}>返回拼团列表</Button>} />;
  }

  // 活动详情
  const detail = query.data;
  // 是否可参与（后端只返回进行中活动；此处用状态兜底）
  const joinable = detail.status === 'ongoing';
  // 待成团团组
  const groups = detail.groups;

  return (
    <div>
      <PageHeader title={detail.name} extra={<StatusTag domain="groupon" status={detail.status} />} />
      <Card style={{ marginBottom: 16 }}>
        <Space direction="vertical" size={8}>
          <Typography.Text type="secondary">{detail.courseName}</Typography.Text>
          <MoneyText cents={detail.priceCents} size="large" />
          <Typography.Text type="secondary">{detail.targetSize} 人成团</Typography.Text>
          <Countdown
            deadlineAt={detail.endAt}
            prefix="距活动结束"
            onExpire={() => void query.refetch()}
          />
          {user ? (
            <ConfirmAction
              title="发起拼团"
              content={`确认以 ${formatCents(detail.priceCents)} 发起新团？支付成功后计入成团人数。`}
              onConfirm={async () => {
                try {
                  await mutation.mutateAsync({ groupId: null });
                } catch {
                  // 错误已由 onError 提示
                }
              }}
            >
              <Button type="primary" disabled={!joinable} loading={mutation.isPending}>
                发起新团
              </Button>
            </ConfirmAction>
          ) : (
            <Button type="primary" onClick={gotoLogin}>发起新团</Button>
          )}
        </Space>
      </Card>

      <Card title="待成团团组">
        {groups.length === 0 ? (
          <EmptyState description="暂无可加入的团，发起新团吧" />
        ) : (
          <List
            dataSource={groups}
            renderItem={(group) => {
              // 团是否已满
              const full = group.memberCount >= group.targetSize;
              // 参团按钮是否禁用
              const disabled = !joinable || full || group.status !== 'pending';
              return (
                <List.Item
                  actions={[
                    user ? (
                      <ConfirmAction
                        key="join"
                        title="参与拼团"
                        content={`确认加入该团？还差 ${group.targetSize - group.memberCount} 人成团。`}
                        onConfirm={async () => {
                          try {
                            await mutation.mutateAsync({ groupId: group.id });
                          } catch {
                            // 错误已由 onError 提示
                          }
                        }}
                      >
                        <Button disabled={disabled} loading={mutation.isPending}>
                          参团
                        </Button>
                      </ConfirmAction>
                    ) : (
                      <Button key="login" onClick={gotoLogin}>参团</Button>
                    ),
                  ]}
                >
                  <Space direction="vertical" size={2}>
                    <Typography.Text>
                      已支付 {group.memberCount}/{group.targetSize} 人
                      {full ? '（已满）' : ''}
                    </Typography.Text>
                    <Countdown
                      deadlineAt={group.expiresAt}
                      prefix="距结束"
                      onExpire={() => void query.refetch()}
                    />
                  </Space>
                </List.Item>
              );
            }}
          />
        )}
      </Card>
    </div>
  );
}
```
