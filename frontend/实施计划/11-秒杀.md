**契约修正（必须先做）**：后端 `flash-sale.mapper.ts` 实际返回 `remaining` 与 `version`，而 `openapi.yaml` 的 `FlashSaleActivityBrief` 未声明，导致生成的类型缺字段。按 README 风险 4「漂移先改 openapi」：先在 `backend/openapi.yaml` 的 `FlashSaleActivityBrief.properties` 中补：

```yaml
        remaining: { type: integer, description: 剩余可购数量（Redis 预占后的服务端值） }
        version: { type: integer, description: 乐观锁版本号（管理端编辑回传） }
```

再执行 `cd frontend && npm run gen:api` 并提交生成物；禁止在 `src/api/types.ts` 手写 `remaining` 绕开契约。

**后端行为约定（已确认）**：
- `GET /flash-sales` 公开，返回 `not_started` 与 `ongoing`，按开始时间正序；`remaining` 为 Redis 预占后的服务端剩余。
- `GET /flash-sales/:id` 公开；不存在为 404（`code=1002`）。
- `POST /flash-sales/:id/purchase` 必带 `Idempotency-Key`，成功 201 返回 `{ order }`（`pending`，需再走支付）。
- 未开始/已结束为 `422 + code=5002`；售罄为 `422 + code=4001`；一人一活动限购一次为 `409 + code=5003`。

// src/features/flash-sale/FlashSaleCard.tsx
import { Card, Space, Tag, Typography } from 'antd';
import { Link } from 'react-router';
import type { FlashSaleActivityBrief } from '@/api/types';
import { Countdown } from '@/components/common/Countdown';
import { MoneyText } from '@/components/common/MoneyText';
import { formatShanghai } from '@/lib/datetime';

// 秒杀活动卡片：秒杀价、剩余库存、起止时间与倒计时
export function FlashSaleCard(props: { activity: FlashSaleActivityBrief; onExpire?: () => void }) {
  // 是否已售罄
  const soldOut = props.activity.remaining <= 0;
  // 未开始的活动倒计时到开始时间，进行中倒计时到结束时间
  const notStarted = props.activity.status === 'not_started';

  return (
    <Card hoverable style={{ height: '100%' }}>
      <Link to={`/flash-sales/${props.activity.id}`}>{props.activity.name}</Link>
      <Typography.Paragraph type="secondary" style={{ marginBottom: 8 }}>
        {props.activity.courseName}
      </Typography.Paragraph>
      <Space direction="vertical" size={4}>
        <MoneyText cents={props.activity.priceCents} />
        {soldOut ? (
          <Tag color="red">已售罄</Tag>
        ) : (
          <Typography.Text type="secondary">剩余 {props.activity.remaining} 件</Typography.Text>
        )}
        <Typography.Text type="secondary">
          {formatShanghai(props.activity.startAt)} ~ {formatShanghai(props.activity.endAt)}
        </Typography.Text>
        <Countdown
          deadlineAt={notStarted ? props.activity.startAt : props.activity.endAt}
          prefix={notStarted ? '距开始' : '距结束'}
          onExpire={props.onExpire}
        />
      </Space>
    </Card>
  );
}

```tsx
// src/features/flash-sale/FlashSaleListPage.tsx
import { useEffect, useState } from 'react';
import { Col, Pagination, Row } from 'antd';
import { useQuery } from '@tanstack/react-query';
import { listFlashSales } from '@/api/promotion';
import { queryKeys } from '@/api/query-keys';
import { EmptyState } from '@/components/common/EmptyState';
import { PageHeader } from '@/components/common/PageHeader';
import { usePagination } from '@/hooks/usePagination';
import { FlashSaleCard } from './FlashSaleCard';

// 秒杀列表每页数量
const PAGE_SIZE = 12;

// 秒杀列表：未开始与进行中活动 + 分页 + 倒计时
export default function FlashSaleListPage() {
  // 当前总数（分页入参）
  const [total, setTotal] = useState(0);
  // 页码与页大小
  const { page, pageSize, onChange } = usePagination(total, PAGE_SIZE);

  // 活动分页查询
  const query = useQuery({
    queryKey: queryKeys.flashSales(page),
    queryFn: () => listFlashSales(page, pageSize),
  });

  // 同步后端总数
  useEffect(() => {
    if (query.data) setTotal(query.data.total);
  }, [query.data]);

  // 倒计时到期：重取列表，切换未开始/已结束展示
  const handleExpire = () => {
    void query.refetch();
  };

  // 当前页活动
  const items = query.data?.items ?? [];

  return (
    <div>
      <PageHeader title="限时秒杀" />
      {items.length === 0 && !query.isLoading ? (
        <EmptyState description="暂无秒杀活动" />
      ) : (
        <Row gutter={[16, 16]}>
          {items.map((item) => (
            <Col key={item.id} xs={24} md={12} lg={8}>
              <FlashSaleCard activity={item} onExpire={handleExpire} />
            </Col>
          ))}
        </Row>
      )}
      <Pagination
        current={page}
        pageSize={pageSize}
        total={total}
        onChange={onChange}
        style={{ marginTop: 16, textAlign: 'right' }}
      />
    </div>
  );
}

  const query = useQuery({
    queryKey: queryKeys.flashSale(activityId),
    queryFn: () => getFlashSale(activityId),
    enabled: Number.isInteger(activityId) && activityId > 0,
  });

  // 失败提示并刷新库存/状态
  const handleError = (error: ApiError) => {
    if (error.code === 4001) {
      message.warning('已售罄');
    } else if (error.code === 5002) {
      message.warning('秒杀活动未开始或已结束');
    } else if (error.code === 5003) {
      message.warning('同一活动限购一次');
    } else {
      message.error(error.message);
    }
    void query.refetch();
  };

  // 抢购：一次用户意图一个幂等键
  const mutation = useIdempotentMutation<{ order: Order }, void>(
    (_vars, key) => purchaseFlashSale(activityId, key),
    {
      onSuccess: ({ order }) => {
        // 库存已变化，失效详情与列表
        void queryClient.invalidateQueries({ queryKey: queryKeys.flashSale(activityId) });
        void queryClient.invalidateQueries({ queryKey: ['flash-sales'] });
        // 进入既有订单支付模式
        navigate(`/checkout/${order.courseId}?orderId=${order.id}`);
      },
      onError: handleError,
    },
  );

  // 游客：跳登录并回跳详情
  const gotoLogin = () => {
    navigate(`/login?redirect=${encodeURIComponent(`/flash-sales/${activityId}`)}`);
  };

  if (query.isLoading) return <Card loading />;
  if (query.error || !query.data) {
    return (
      <EmptyState
        description="秒杀活动不存在"
        action={<Button onClick={() => navigate('/flash-sales')}>返回秒杀列表</Button>}
      />
    );
  }

  // 活动详情
  const sale = query.data;
  // 是否未开始
  const notStarted = sale.status === 'not_started';
  // 是否售罄
  const soldOut = sale.remaining <= 0;
  // 是否可抢购
  const purchasable = sale.status === 'ongoing' && !soldOut;
  // 按钮文案：售罄优先，其次未开始/已结束
  const buttonLabel = soldOut
    ? '已售罄'
    : notStarted
      ? '未开始'
      : sale.status !== 'ongoing'
        ? '已结束'
        : '立即抢购';

  return (
    <div>
      <PageHeader title={sale.name} extra={<StatusTag domain="flash-sale" status={sale.status} />} />
      <Card>
        <Space direction="vertical" size={8}>
          <Typography.Text type="secondary">{sale.courseName}</Typography.Text>
          <MoneyText cents={sale.priceCents} size="large" />
          <Typography.Text type="secondary">
            总库存 {sale.stock}，已售 {sale.sold}，剩余 {sale.remaining}
          </Typography.Text>
          <Countdown
            deadlineAt={notStarted ? sale.startAt : sale.endAt}
            prefix={notStarted ? '距开始' : '距结束'}
            onExpire={() => void query.refetch()}
          />
          {!purchasable ? (
            <Button type="primary" disabled>{buttonLabel}</Button>
          ) : user ? (
            <ConfirmAction
              title="确认抢购"
              content="抢购会创建待支付订单，请在支付结果页完成支付。确认继续？"
              onConfirm={async () => {
                try {
                  await mutation.mutateAsync();
                } catch {
                  // 错误已由 onError 提示
                }
              }}
            >
              <Button type="primary" loading={mutation.isPending}>立即抢购</Button>
            </ConfirmAction>
          ) : (
            <Button type="primary" onClick={gotoLogin}>立即抢购</Button>
          )}
        </Space>
      </Card>
    </div>
  );
}
```
