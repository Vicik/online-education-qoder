**后端行为约定（已确认）**：
- `GET /posts` 与 `GET /posts/:id` 公开；登录用户返回 `likedByMe`，游客不返回该字段。
- 发帖、一级评论、回复均必带 `Idempotency-Key`（8–128）；点赞/取消点赞无需幂等键。
- 命中屏蔽词为 `422 + code=7001`，`data` 为命中的词数组（包含匹配、忽略英文大小写）。
- 回复只能挂在一级评论下（不产生第三层）；`replyToUserId` 决定「回复 @昵称」。
- 删除一级评论由数据库级联删除其回复；删除他人内容为 `403 + code=2002`。

```ts
// src/api/community.ts
import { request } from './client';
import type { ApiError } from './errors';
import type {
  CommentView,
  PageResult,
  PostBrief,
  PostDetail,
  ReplyView,
} from './types';

// 信息流（公开；登录用户附 likedByMe）
export function listPosts(page = 1, pageSize = 10): Promise<PageResult<PostBrief>> {
  return request({ url: '/posts', method: 'GET', params: { page, pageSize } });
}

// 帖子详情（公开；两层评论）
export function getPost(postId: number): Promise<PostDetail> {
  return request({ url: `/posts/${postId}`, method: 'GET' });
}

// 发帖：富文本内容 + 幂等键
export function createPost(
  payload: { content: string },
  idempotencyKey: string,
): Promise<PostBrief> {
  return request({
    url: '/posts',
    method: 'POST',
    data: payload,
    headers: { 'Idempotency-Key': idempotencyKey },
  });
}

// 删除自己的帖子（评论由后端级联删除）
export function deletePost(postId: number): Promise<void> {
  return request({ url: `/posts/${postId}`, method: 'DELETE' });
}

// 点赞（幂等，无需幂等键）
export function likePost(postId: number): Promise<void> {
  return request({ url: `/posts/${postId}/like`, method: 'POST', data: {} });
}

// 取消点赞（幂等）
export function unlikePost(postId: number): Promise<void> {
  return request({ url: `/posts/${postId}/like`, method: 'DELETE' });
}

// 发表一级评论：幂等键
export function createComment(
  postId: number,
  payload: { content: string },
  idempotencyKey: string,
): Promise<CommentView> {
  return request({
    url: `/posts/${postId}/comments`,
    method: 'POST',
    data: payload,
    headers: { 'Idempotency-Key': idempotencyKey },
  });
}

// 回复一级评论：不产生第三层，replyToUserId 表达回复目标
export function createReply(
  commentId: number,
  payload: { content: string; replyToUserId: number | null },
  idempotencyKey: string,
): Promise<ReplyView> {
  return request({
    url: `/comments/${commentId}/replies`,
    method: 'POST',
    data: payload,
    headers: { 'Idempotency-Key': idempotencyKey },
  });
}

// 删除自己的评论（一级评论级联删除回复）
export function deleteComment(commentId: number): Promise<void> {
  return request({ url: `/comments/${commentId}`, method: 'DELETE' });
}

// 我的帖子（仅自己发布）
export function listMyPosts(page = 1, pageSize = 10): Promise<PageResult<PostBrief>> {
  return request({ url: '/user/posts', method: 'GET', params: { page, pageSize } });
}

// 从 7001 错误中提取命中词：后端 data 为字符串数组
export function extractBlockedWords(error: ApiError): string[] {
  return Array.isArray(error.data)
    ? error.data.filter((word): word is string => typeof word === 'string')
    : [];
}
```

```tsx
// src/features/community/LikeButton.tsx
import { useEffect, useState } from 'react';
import { Button, message } from 'antd';
import { useLocation, useNavigate } from 'react-router';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { likePost, unlikePost } from '@/api/community';
import type { PostBrief } from '@/api/types';
import { usePermission } from '@/hooks/usePermission';

// 点赞按钮：游客登录回跳；登录用户按当前状态点赞/取消
export function LikeButton(props: { post: PostBrief }) {
  // 当前用户
  const { user } = usePermission();
  // 路由跳转
  const navigate = useNavigate();
  // 当前地址（登录回跳）
  const location = useLocation();
  // 查询缓存
  const queryClient = useQueryClient();
  // 本地点赞态（服务端返回后同步）
  const [liked, setLiked] = useState(props.post.likedByMe ?? false);
  // 本地点赞数
  const [count, setCount] = useState(props.post.likeCount);

  // 列表/详情刷新后同步本地状态
  useEffect(() => {
    setLiked(props.post.likedByMe ?? false);
    setCount(props.post.likeCount);
  }, [props.post]);

  // 点赞/取消点赞
  const mutation = useMutation({
    mutationFn: () => (liked ? unlikePost(props.post.id) : likePost(props.post.id)),
    onSuccess: () => {
      // 本地即时反馈
      setLiked((prev) => !prev);
      setCount((prev) => prev + (liked ? -1 : 1));
      // 刷新信息流与详情
      void queryClient.invalidateQueries({ queryKey: ['posts'] });
      void queryClient.invalidateQueries({ queryKey: ['post', props.post.id] });
    },
    onError: (error: Error) => {
      message.error(error.message);
    },
  });

  // 游客：跳登录并回跳当前地址
  if (!user) {
    return (
      <Button
        type="link"
        onClick={() =>
          navigate(`/login?redirect=${encodeURIComponent(location.pathname + location.search)}`)
        }
      >
        点赞 {count}
      </Button>
    );
  }

  return (
    <Button
      type="link"
      loading={mutation.isPending}
      onClick={() => mutation.mutate()}
    >
      {liked ? '已点赞' : '点赞'} {count}
    </Button>
  );
}
```

```tsx
// src/features/community/FeedPage.tsx
import { useEffect, useState } from 'react';
import { Button, Pagination } from 'antd';
import { Link, useNavigate } from 'react-router';
import { useQuery } from '@tanstack/react-query';
import { listPosts } from '@/api/community';
import { queryKeys } from '@/api/query-keys';
import { EmptyState } from '@/components/common/EmptyState';
import { PageHeader } from '@/components/common/PageHeader';
import { usePagination } from '@/hooks/usePagination';
import { usePermission } from '@/hooks/usePermission';
import { PostCard } from './PostCard';

// 信息流每页数量
const PAGE_SIZE = 10;

// 社区信息流：分页、发帖入口、游客登录回跳
export default function FeedPage() {
  // 当前用户
  const { user } = usePermission();
  // 路由跳转
  const navigate = useNavigate();
  // 当前总数（分页入参）
  const [total, setTotal] = useState(0);
  // 页码与页大小
  const { page, pageSize, onChange } = usePagination(total, PAGE_SIZE);

  // 信息流查询
  const query = useQuery({
    queryKey: queryKeys.posts(page),
    queryFn: () => listPosts(page, pageSize),
  });

  // 同步后端总数
  useEffect(() => {
    if (query.data) setTotal(query.data.total);
  }, [query.data]);

  // 发帖入口：游客先登录
  const gotoNewPost = () => {
    if (!user) {
      navigate('/login?redirect=%2Fcommunity%2Fnew');
      return;
    }
    navigate('/community/new');
  };

  // 当前页帖子
  const items = query.data?.items ?? [];

  return (
    <div>
      <PageHeader
        title="学习社区"
        extra={
          <Button type="primary" onClick={gotoNewPost}>发布帖子</Button>
        }
      />
      {items.length === 0 && !query.isLoading ? (
        <EmptyState description="还没有帖子，来发第一帖吧" />
      ) : (
        items.map((item) => <PostCard key={item.id} post={item} />)
      )}
      <Pagination
        current={page}
        pageSize={pageSize}
        total={total}
        onChange={onChange}
        style={{ textAlign: 'right' }}
      />
    </div>
  );
}
```

```tsx
// src/features/community/CommentTree.tsx
import { useState } from 'react';
import { Button, Card, Space, Typography, message } from 'antd';
import { useLocation, useNavigate } from 'react-router';
import { useQueryClient } from '@tanstack/react-query';
import { createReply, deleteComment, extractBlockedWords } from '@/api/community';
import type { ApiError } from '@/api/errors';
import type { PostDetail, ReplyView } from '@/api/types';
import { ConfirmAction } from '@/components/common/ConfirmAction';
import { RichTextViewer } from '@/components/common/RichTextViewer';
import { useIdempotentMutation } from '@/hooks/useIdempotentMutation';
import { usePermission } from '@/hooks/usePermission';
import { formatShanghai } from '@/lib/datetime';
import { CommentForm } from './CommentForm';

// 两级评论树：一级评论 + 回复；回复目标显示「回复 @昵称」
export function CommentTree(props: { post: PostDetail }) {
  // 当前用户
  const { user } = usePermission();
  // 路由跳转
  const navigate = useNavigate();
  // 当前地址（登录回跳）
  const location = useLocation();
  // 查询缓存
  const queryClient = useQueryClient();
  // 正在回复的一级评论 id
  const [replyingTo, setReplyingTo] = useState<number | null>(null);

  // 游客跳登录
  const requireLogin = () => {
    navigate(`/login?redirect=${encodeURIComponent(location.pathname + location.search)}`);
  };

  // 刷新详情
  const refresh = () => {
    void queryClient.invalidateQueries({ queryKey: ['post', props.post.id] });
    void queryClient.invalidateQueries({ queryKey: ['posts'] });
  };

  // 删除评论（仅本人入口可见）
  const handleDelete = async (commentId: number) => {
    try {
      await deleteComment(commentId);
      message.success('已删除');
      refresh();
    } catch (error) {
      message.error((error as ApiError).message);
    }
  };

  // 回复一级评论：一次用户意图一个幂等键
  const replyMutation = useIdempotentMutation<
    ReplyView,
    { commentId: number; replyToUserId: number; content: string }
  >(
    (vars, key) =>
      createReply(
        vars.commentId,
        { content: vars.content, replyToUserId: vars.replyToUserId },
        key,
      ),
    {
      onSuccess: () => {
        message.success('回复成功');
        setReplyingTo(null);
        refresh();
      },
      onError: (error) => {
        const words = extractBlockedWords(error);
        message.error(
          words.length > 0 ? `内容包含屏蔽词：${words.join('、')}` : error.message,
        );
      },
    },
  );

  // 提交回复：成功返回 true，由 CommentForm 清空输入
  const handleReply = async (
    commentId: number,
    replyToUserId: number,
    content: string,
  ): Promise<boolean> => {
    try {
      await replyMutation.mutateAsync({ commentId, replyToUserId, content });
      return true;
    } catch {
      return false;
    }
  };

  // 一级评论列表
  const comments = props.post.comments;

  return (
    <div>
      {comments.map((comment) => (
        <Card key={comment.id} size="small" style={{ marginBottom: 12 }}>
          <Space direction="vertical" style={{ width: '100%' }}>
            <Space>
              <Typography.Text strong>{comment.nickname}</Typography.Text>
              <Typography.Text type="secondary">{formatShanghai(comment.createdAt)}</Typography.Text>
            </Space>
            <RichTextViewer html={comment.content} />
            <Space>
              <Button type="link" onClick={() => (user ? setReplyingTo(comment.id) : requireLogin())}>
                回复
              </Button>
              {user?.id === comment.userId && (
                <ConfirmAction
                  title="删除评论"
                  content="删除一级评论会同时删除其全部回复，确认删除？"
                  danger
                  onConfirm={() => handleDelete(comment.id)}
                >
                  <Button type="link" danger>删除评论</Button>
                </ConfirmAction>
              )}
            </Space>

            {comment.replies.map((reply) => (
              <Card key={reply.id} size="small" type="inner">
                <Space direction="vertical" style={{ width: '100%' }}>
                  <Space>
                    <Typography.Text strong>{reply.nickname}</Typography.Text>
                    {reply.replyToNickname && (
                      <Typography.Text type="secondary">回复 @{reply.replyToNickname}</Typography.Text>
                    )}
                    <Typography.Text type="secondary">{formatShanghai(reply.createdAt)}</Typography.Text>
                  </Space>
                  <RichTextViewer html={reply.content} />
                </Space>
              </Card>
            ))}

            {replyingTo === comment.id && (
              <CommentForm
                placeholder={`回复 @${comment.nickname}`}
                submitting={replyMutation.isPending}
                onSubmit={(content) => handleReply(comment.id, comment.userId, content)}
              />
            )}
          </Space>
        </Card>
      ))}
    </div>
  );
}
```

```tsx
    queryKey: queryKeys.post(postId),
    queryFn: () => getPost(postId),
    enabled: Number.isInteger(postId) && postId > 0,
  });

      onSuccess: () => {
        message.success('评论成功');
        void queryClient.invalidateQueries({ queryKey: queryKeys.post(postId) });
        void queryClient.invalidateQueries({ queryKey: ['posts'] });
      },
      onError: (error) => {
        const words = extractBlockedWords(error);
        message.error(words.length > 0 ? `内容包含屏蔽词：${words.join('、')}` : error.message);
      },
    },
  );

    try {
      await deletePost(postId);
      message.success('已删除');
      void queryClient.invalidateQueries({ queryKey: ['posts'] });

      message.error((error as ApiError).message);

  const gotoLogin = () => {
    navigate(`/login?redirect=${encodeURIComponent(location.pathname + location.search)}`);

      <EmptyState
        description="帖子不存在"
        action={<Button onClick={() => navigate('/community')}>返回社区</Button>}
      />
    );
  }

  // 帖子详情
  const post = query.data;
  // 是否本人帖子
  const isOwner = user?.id === post.userId;

  return (
    <div>
      <PageHeader
        title="帖子详情"
        extra={
          isOwner ? (
            <ConfirmAction
              title="删除帖子"
              content="删除后帖子与全部评论不可恢复，确认删除？"
              danger
              onConfirm={handleDeletePost}
            >
              <Button danger>删除帖子</Button>
            </ConfirmAction>

          <Space>
            <Typography.Text strong>{post.nickname}</Typography.Text>
            <Typography.Text type="secondary">{formatShanghai(post.createdAt)}</Typography.Text>

            <CommentForm
              placeholder="写下你的评论"
              submitting={commentMutation.isPending}

                  await commentMutation.mutateAsync({ content });
                  return true;
                } catch {
                  return false;
                }
              }}
            />
          </div>
        ) : (
          <Button type="link" onClick={gotoLogin}>登录后发表评论</Button>
        )}
        <CommentTree post={post} />
      </Card>
    </div>
  );
}
```
