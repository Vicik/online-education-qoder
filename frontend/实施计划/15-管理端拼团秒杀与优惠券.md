**后端行为约定（已确认）**：
- 三组接口全部仅 `admin` 可访问；游客 401，普通用户 403（`code=2002`）。
- 拼团/秒杀列表返回全部状态，按创建时间倒序，**没有状态筛选参数**；优惠券列表支持 `status=disabled|enabled`。
- 拼团创建：仅已发布课程、`targetSize ≥ 2`、结束晚于开始、同课程活动时间不得与拼团/秒杀重叠（服务端 422）；新活动 `not_started`。
- 秒杀创建：仅已发布课程、`stock ≥ 1`、时间合法、时间不重叠；新活动 `not_started`。
- 拼团/秒杀编辑：仅 `not_started`，`version` 必填且必须匹配，否则 409 + `code=4004`；秒杀编辑库存不得小于已售。
- 取消：仅 `not_started`；openapi 未声明请求体，前端不传 `version`，并发状态变化由服务端 422/409 兜底。
- 优惠券创建：默认 `disabled`；`enableOnCreate=true` 时创建即启用；`perUserLimit` 服务端强制为 1；适用课程必须全部已发布且非空。
- 优惠券编辑：仅 `disabled` 可改，`version` 必填；`enable` 仅停用态可执行；`disable` 仅启用态可执行，并把 `unused`/`restored` 领取记录永久置为 `invalid`（不可恢复）。
- 409 统一提示「数据已被修改，请刷新后重试」；前端弹窗提供「刷新」，不自动覆盖。
- 管理端创建类 POST 的前端仍发送 `Idempotency-Key`（设计框架 §6.3）；后端当前未强制，重复提交风险由服务端状态校验兜底。

import { request } from '../client';
import type {

  return request({

    headers: { 'Idempotency-Key': idempotencyKey },

}

    data: input,

import dayjs, { type Dayjs } from 'dayjs';

// 服务端 ISO → DatePicker 值：以上海时区钟面展示
export function toPickerValue(iso: string): Dayjs {
  return dayjs.utc(iso).tz('Asia/Shanghai');

// DatePicker 值 → 服务端 ISO：钟面时间按上海时区解释，再转 UTC
export function fromPickerValue(value: Dayjs): string {

// src/features/admin/groupons/GrouponFormDrawer.tsx
import { App as AntdApp, Button, DatePicker, Drawer, Form, Input, InputNumber } from 'antd';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import type { Dayjs } from 'dayjs';
import { createGroupon, updateGroupon } from '@/api/admin/groupons';
import type { ApiError } from '@/api/errors';
import type { AdminGroupon } from '@/api/types';
import { useIdempotentMutation } from '@/hooks/useIdempotentMutation';
import { CourseSelect } from '../promotions/shared';
import { toPickerValue } from '../promotions/time';
import {
  toGrouponCreatePayload,
  toGrouponUpdatePayload,
  type GrouponFormValues,
} from './payload';

// 抽屉属性：activity 为空表示创建
export function GrouponFormDrawer(props: {
  open: boolean;
  activity: AdminGroupon | null;
  onClose: () => void;
  onSaved: () => void;
  onConflict: () => void;
}) {
  // AntD 消息与确认框
  const { modal, message } = AntdApp.useApp();
  // 查询客户端：写操作后刷新列表
  const queryClient = useQueryClient();
  // 表单实例
  const [form] = Form.useForm<GrouponFormValues>();
  // 是否编辑模式
  const isEdit = props.activity !== null;

  // 创建：一次用户意图一个幂等键
  const create = useIdempotentMutation<AdminGroupon, GrouponFormValues>({
    mutationFn: (values, idempotencyKey) =>
      createGroupon(toGrouponCreatePayload(values), idempotencyKey),
    onSuccess: async () => {
      await queryClient.invalidateQueries({ queryKey: ['admin', 'groupons'] });
      message.success('拼团活动已创建');
      props.onSaved();
    },
    onError: (error) => message.error(error.message),
  });

  // 编辑：PUT 携带 version；409 交给父级刷新
  const update = useMutation<AdminGroupon, ApiError, GrouponFormValues>({
    mutationFn: (values) =>
      updateGroupon(
        props.activity!.id,
        toGrouponUpdatePayload(values, props.activity!.version),
      ),
    onSuccess: async () => {
      await queryClient.invalidateQueries({ queryKey: ['admin', 'groupons'] });
      message.success('拼团活动已保存');
      props.onSaved();
    },
    onError: (error) => {
      if (error.status === 409) {
        props.onConflict();
        return;
      }
      message.error(error.message);
    },
  });

  // 表单初始值：编辑回填，创建默认 2 人成团
  const initialValues = props.activity
    ? {
        courseId: props.activity.courseId,
        name: props.activity.name,
        priceCents: props.activity.priceCents,
        targetSize: props.activity.targetSize,
        startAt: toPickerValue(props.activity.startAt),
        endAt: toPickerValue(props.activity.endAt),
      }
    : { targetSize: 2 };

  // 校验通过后二次确认再提交
  function handleFinish(values: GrouponFormValues) {
    modal.confirm({
      title: isEdit ? '保存拼团活动' : '创建拼团活动',
      content: isEdit ? '确认保存本次活动修改？' : '确认创建该拼团活动？',
      okText: isEdit ? '确认保存' : '确认创建',
      cancelText: '取消',
      onOk: async () => {
        try {
          await (isEdit
            ? update.mutateAsync(values)
            : create.mutateAsync(values));
        } catch (error) {
          if ((error as ApiError).status === 409) return;
          throw error;
        }
      },
    });
  }

  return (
    <Drawer
      title={isEdit ? '编辑拼团活动' : '创建拼团活动'}
      width={520}
      open={props.open}
      onClose={props.onClose}
      destroyOnClose
    >
      <Form
        key={props.activity?.id ?? 'create'}
        form={form}
        layout="vertical"
        initialValues={initialValues}
        onFinish={handleFinish}
      >
        <Form.Item
          label="关联课程"
          name="courseId"
          rules={[{ required: true, message: '请选择关联课程' }]}
        >
          <CourseSelect disabled={isEdit} />
        </Form.Item>
        <Form.Item
          label="活动名称"
          name="name"
          rules={[{ required: true, whitespace: true, message: '请填写活动名称' }]}
        >
          <Input maxLength={40} />
        </Form.Item>
        <Form.Item
          label="拼团价"
          name="priceCents"
          rules={[{ required: true, message: '请填写拼团价' }]}
        >
          <MoneyInput placeholder="拼团价（元）" />
        </Form.Item>
        <Form.Item
          label="成团人数"
          name="targetSize"
          rules={[{ required: true, message: '请填写成团人数' }]}
        >
          <InputNumber min={2} precision={0} style={{ width: '100%' }} />
        </Form.Item>
        <Form.Item
          label="开始时间"
          name="startAt"
          rules={[{ required: true, message: '请选择开始时间' }]}
        >
          <DatePicker
            showTime
            format="YYYY-MM-DD HH:mm:ss"
            placeholder="开始时间"
            style={{ width: '100%' }}
          />
        </Form.Item>
        <Form.Item
          label="结束时间"
          name="endAt"
          dependencies={['startAt']}
          rules={[
            { required: true, message: '请选择结束时间' },
            ({ getFieldValue }) => ({
              validator(_rule, value: Dayjs | null) {
                // 开始时间
                const startAt = getFieldValue('startAt') as Dayjs | null;
                if (!value || !startAt || value.valueOf() > startAt.valueOf()) {
                  return Promise.resolve();
                }
                return Promise.reject(new Error('结束时间必须晚于开始时间'));

            showTime
            format="YYYY-MM-DD HH:mm:ss"
            placeholder="结束时间"
            style={{ width: '100%' }}
          />

          type="primary"
          htmlType="submit"
          block
          loading={create.isPending || update.isPending}

import { useEffect, useState } from 'react';
import { App as AntdApp, Button, Space } from 'antd';

import type { ApiError } from '@/api/errors';
import { queryKeys } from '@/api/query-keys';
import type { AdminGroupon } from '@/api/types';
import { AppTable } from '@/components/common/AppTable';
import { ConfirmAction } from '@/components/common/ConfirmAction';

// 列表每页数量
const PAGE_SIZE = 10;

// 拼团管理：列表、抽屉创建/编辑、取消
export default function GrouponListPage() {
  // 查询客户端
  const queryClient = useQueryClient();
  // AntD 消息与确认框
  const { modal, message } = AntdApp.useApp();
  // 当前总数（分页入参）
  const [total, setTotal] = useState(0);
  // 页码与页大小
  const { page, pageSize, onChange } = usePagination(total, PAGE_SIZE);
  // 抽屉开关
  const [drawerOpen, setDrawerOpen] = useState(false);
  // 正在编辑的活动（空表示创建）
  const [editing, setEditing] = useState<AdminGroupon | null>(null);

  // 活动列表查询
  const query = useQuery({
    queryKey: queryKeys.adminGroupons(page),
    queryFn: () => listAdminGroupons(page, pageSize),
  });

  // 同步后端总数
  useEffect(() => {
    if (query.data) setTotal(query.data.total);
  }, [query.data]);

  // 409 刷新弹窗：只重取列表，不覆盖他人修改
  function showConflict(onRefresh?: () => void) {
    modal.confirm({
      title: '内容已在其他页面被修改',
      content: '请刷新后重试，系统不会自动覆盖他人修改。',
      okText: '刷新',
      cancelText: '取消',
      onOk: async () => {
        await queryClient.invalidateQueries({ queryKey: ['admin', 'groupons'] });
        onRefresh?.();
      },
    });
  }

  // 取消活动：仅未开始；并发状态变化由服务端校验
  const cancelMutation = useMutation<AdminGroupon, ApiError, number>({
    mutationFn: cancelGroupon,
    onSuccess: async () => {
      await queryClient.invalidateQueries({ queryKey: ['admin', 'groupons'] });
      message.success('活动已取消');
    },
    onError: (error) => {
      if (error.status === 409) {
        showConflict();
        return;
      }
      message.error(error.message);
    },
  });

  // 打开创建抽屉
  function openCreate() {
    setEditing(null);
    setDrawerOpen(true);
  }

  // 打开编辑抽屉
  function openEdit(activity: AdminGroupon) {
    setEditing(activity);
    setDrawerOpen(true);
  }

  // 关闭抽屉并清空编辑对象
  function closeDrawer() {
    setDrawerOpen(false);
    setEditing(null);
  }

  // 表格列
  const columns = [
    { title: '活动名称', dataIndex: 'name', key: 'name' },
    { title: '课程', dataIndex: 'courseName', key: 'courseName' },
    {
      title: '拼团价',
      dataIndex: 'priceCents',
      key: 'priceCents',
      render: (value: number) => <MoneyText cents={value} />,
    },
    { title: '成团人数', dataIndex: 'targetSize', key: 'targetSize' },
    {
      title: '活动时间',
      key: 'period',
      render: (_: unknown, record: AdminGroupon) =>
        `${formatShanghai(record.startAt)} ~ ${formatShanghai(record.endAt)}`,
    },
    {
      title: '状态',
      dataIndex: 'status',
      key: 'status',
      render: (value: AdminGroupon['status']) => (
        <StatusTag domain="groupon" status={value} />
      ),
    },
    {
      title: '操作',
      key: 'actions',
      render: (_: unknown, record: AdminGroupon) => {
        // 仅未开始可编辑/取消
        const editable = record.status === 'not_started';
        return (
          <Space>
            <Button
              type="link"
              disabled={!editable}
              onClick={() => openEdit(record)}
            >
              编辑
            </Button>
            <ConfirmAction
              title="取消拼团活动"
              content={`确认取消「${record.name}」？取消后不可恢复。`}
              okText="确认取消"
              danger
              disabled={!editable}
              onConfirm={async () => {
                await cancelMutation.mutateAsync(record.id);
              }}
            >
              <Button type="link" danger disabled={!editable}>
                取消
              </Button>
            </ConfirmAction>
          </Space>
        );
      },
    },
  ];

  return (
    <div>
      <PageHeader
        title="拼团管理"
        extra={
          <Button type="primary" onClick={openCreate}>
            创建拼团
          </Button>
        }
      />
      <AppTable
        columns={columns}
        dataSource={query.data?.items ?? []}
        rowKey="id"
        total={total}
        page={page}
        pageSize={pageSize}
        loading={query.isLoading}
        error={query.error}
        onRetry={() => void query.refetch()}
        onPageChange={onChange}
        emptyText="还没有拼团活动"
      />
      <GrouponFormDrawer
        open={drawerOpen}
        activity={editing}
        onClose={closeDrawer}
        onSaved={closeDrawer}
        onConflict={() => {
          closeDrawer();
          showConflict();
        }}
      />
    </div>
  );
}

## 已知边界

- **拼团/秒杀没有详情接口与编辑路由**：编辑直接使用列表行数据。列表可能短暂过期，提交时由 `version` 409 与服务端状态 422 兜底；收到 409 先关闭抽屉再刷新列表，避免继续用旧版本提交。
- **拼团/秒杀列表没有状态筛选**：后端未提供参数，前端不伪造本地筛选；优惠券列表有 `status` 筛选。
- **取消/启用/停用不接受 `version`**：openapi 未声明请求体，前端不发送。并发状态变化返回 422 或 409；前端对 409 的处理与编辑一致，对 422 直接提示后端文案。
- **管理端创建类 POST 的幂等键后端未强制**：前端按设计框架 §6.3 一律发送；若后端将来启用校验，前端无需改动。
- **已发布课程下拉最多 100 条**：`pageSize` 上限 100，超过 100 门已发布课程时无法在单个下拉中选全；需后端提供课程搜索参数后再扩展，不在前端拼接分页。
- **活动时间重叠只在服务端校验**：前端不做跨活动查询，创建/编辑时若重叠，展示后端 422 文案并保持表单内容。
- **时间按 Asia/Shanghai 钟面解释**：管理端录入的时间即上海时间，提交前转 UTC；展示统一 `formatShanghai`。跨时区运营人员看到与录入的都是上海时间。
- **停用优惠券的级联不可逆**：`unused`/`restored` 记录被置为 `invalid` 后即使重新启用也不会恢复；前端只做警告与二次确认，不在本地推算受影响数量。
- **`version` 字段对用户端可见**：`CouponTemplateBrief`、`GrouponActivityBrief` 是用户端与管理端共用的 schema，补字段后用户端也会收到；用户端页面不使用该字段，不构成契约破坏。
